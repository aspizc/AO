El operador necesita verificar que sus registries son validos sin arrancar el Gateway entero, y sin un IDE. Esto:

1. Cierra el ciclo Stage A→B: ahora hay un comando real (no stub).
2. Da al CI futuro un check rapido (`agent-run policy validate` → exit 0).
3. Habilita iteracion segura: si el operador edita `policies/*.json`, este comando le dice de inmediato si rompio algo.

### 2. Script Node helper

`gateway/scripts/validate-registries.mjs`:

```javascript
#!/usr/bin/env node
import path from "node:path";
import { fileURLToPath } from "node:url";
import { loadRegistries, RegistryError } from "../src/core/registry.js";

const __filename = fileURLToPath(import.meta.url);
const REPO_ROOT = path.resolve(path.dirname(__filename), "..", "..");
const policiesDir = process.env.AGENTS_POLICIES_DIR
  ? path.resolve(process.env.AGENTS_POLICIES_DIR)
  : path.join(REPO_ROOT, "policies");

try {
  const r = loadRegistries({ policiesDir });
  const out = {
    ok: true,
    policiesDir,
    counts: {
      agents: Object.keys(r.raw().agents).length,
      repositories: Object.keys(r.raw().repositories).length,
      roles: Object.keys(r.raw().roles).length
    }
  };
  process.stdout.write(JSON.stringify(out));
  process.exit(0);
} catch (err) {
  const out = {
    ok: false,
    code: err instanceof RegistryError ? err.code : "UNKNOWN",
    message: err.message,
    details: err.details || null
  };
  process.stdout.write(JSON.stringify(out));
  process.exit(1);
}
```

### 3. Implementacion CLI

Reemplazar el stub en `cli/src/agents_cli/main.py` (subcomando `policy validate`):

```python
import json
import os
import shutil
import subprocess
from pathlib import Path

import typer
from rich.console import Console

console = Console()

REPO_ROOT = Path(__file__).resolve().parents[3]
VALIDATOR = REPO_ROOT / "gateway" / "scripts" / "validate-registries.mjs"


@policy_app.command("validate")
def policy_validate(
    policies_dir: Path = typer.Option(
        None, "--policies-dir",
        help="Defaults to AGENTS_POLICIES_DIR or repo policies/"),
    output_json: bool = typer.Option(False, "--json", help="Machine-readable output"),
) -> None:
    """Validate registries (agent-capabilities, repositories, roles)."""
    if shutil.which("node") is None:
        typer.echo("error: 'node' is not on PATH", err=True)
        raise typer.Exit(code=2)
    if not VALIDATOR.is_file():
        typer.echo(f"error: validator not found at {VALIDATOR}", err=True)
        raise typer.Exit(code=2)

    env = os.environ.copy()
    if policies_dir:
        env["AGENTS_POLICIES_DIR"] = str(policies_dir)

    proc = subprocess.run(
        ["node", str(VALIDATOR)],
        capture_output=True, text=True, env=env, check=False,
    )
    out = proc.stdout.strip()
    try:
        data = json.loads(out)
    except json.JSONDecodeError:
        typer.echo(out, err=True)
        raise typer.Exit(code=proc.returncode or 1)

    if output_json:
        typer.echo(json.dumps(data))
        raise typer.Exit(code=0 if data.get("ok") else 1)

    if data.get("ok"):
        c = data["counts"]
        console.print(f"[green]OK[/green] policies dir: {data['policiesDir']}")
        console.print(f"agents: {c['agents']}  repos: {c['repositories']}  roles: {c['roles']}")
        raise typer.Exit(code=0)
    else:
        console.print(f"[red]FAIL[/red] {data.get('code')}: {data.get('message')}")
        raise typer.Exit(code=1)
```

> Nota: importa `policy_app` y `app` ya existentes; no dupliques el Typer app.

### 4. Test pytest

`tests/cli/test_policy_validate.py`:

```python
import os
import shutil
import subprocess
from pathlib import Path

import pytest
from typer.testing import CliRunner

from agents_cli.main import app

REPO = Path(__file__).resolve().parents[2]
runner = CliRunner()


def test_policy_validate_success():
    if shutil.which("node") is None:
        pytest.skip("node not available")
    result = runner.invoke(app, ["policy", "validate"])
    assert result.exit_code == 0
    assert "OK" in result.stdout


def test_policy_validate_failure_for_missing_registry(tmp_path):
    if shutil.which("node") is None:
        pytest.skip("node not available")
    # tmp dir lacks the registry files
    env_arg = ["--policies-dir", str(tmp_path)]
    result = runner.invoke(app, ["policy", "validate", *env_arg])
    assert result.exit_code != 0
    assert "FAIL" in result.stdout or "FAIL" in result.output


def test_policy_validate_json_mode_emits_valid_json():
    if shutil.which("node") is None:
        pytest.skip("node not available")
    result = runner.invoke(app, ["policy", "validate", "--json"])
    assert result.exit_code == 0
    import json
    parsed = json.loads(result.stdout)
    assert parsed.get("ok") is True
```
