## ¿Por que esta tarea existe?

Tres registries planos no son utilizables directamente: hay que cargar, validar (formato + invariantes cruzadas) y exponer getters tipados. Tambien es la frontera donde queremos **fail-fast**: si un registry esta corrupto, el Gateway no debe arrancar.

Errores tipados (no genericos) son criticos: en Stage E queremos que `agent-run policy validate` muestre al operador exactamente que registry y que campo fallaron.

## ¿Que hay que hacer?

Implementar `gateway/src/core/registry.js` con `loadRegistries({ policiesDir })` que:

1. Lee `agent-capabilities.json`, `repositories.json`, `roles.json`.
2. Valida que existen y son JSON.
3. Valida invariantes cruzadas (ej. `restricted` repos solo permiten agentes con `restricted` en `allowedClassifications`).
4. Devuelve un objeto con getters: `getAgent`, `getRepo`, `getRole`, `getProtectedBranches`.
5. Lanza `RegistryError` con codigo + path + campo.

### 2. `gateway/src/core/registry.js`

```javascript
import fs from "node:fs";
import path from "node:path";

export class RegistryError extends Error {
  constructor(code, message, details = {}) {
    super(message);
    this.name = "RegistryError";
    this.code = code;
    this.details = details;
  }
}

function readJson(absPath, file) {
  if (!fs.existsSync(absPath)) {
    throw new RegistryError("REGISTRY_MISSING", `missing registry file: ${file}`, { absPath });
  }
  try {
    return JSON.parse(fs.readFileSync(absPath, "utf-8"));
  } catch (err) {
    throw new RegistryError("REGISTRY_INVALID_JSON",
      `invalid JSON in ${file}: ${err.message}`, { absPath });
  }
}

function validateAgents(agents) {
  if (typeof agents !== "object" || agents === null) {
    throw new RegistryError("REGISTRY_INVALID_SHAPE",
      "agent-capabilities.json: 'agents' must be an object");
  }
  for (const [id, a] of Object.entries(agents)) {
    if (!Array.isArray(a.allowedClassifications)) {
      throw new RegistryError("REGISTRY_INVALID_AGENT",
        `agent ${id}: allowedClassifications must be an array`);
    }
    if (!Array.isArray(a.allowedRoles)) {
      throw new RegistryError("REGISTRY_INVALID_AGENT",
        `agent ${id}: allowedRoles must be an array`);
    }
  }
}

function validateRepos(repos) {
  const valid = new Set(["unrestricted", "internal", "restricted"]);
  for (const [id, r] of Object.entries(repos)) {
    if (!valid.has(r.classification)) {
      throw new RegistryError("REGISTRY_INVALID_REPO",
        `repo ${id}: invalid classification ${r.classification}`);
    }
    if (!Array.isArray(r.allowedAgents) || r.allowedAgents.length === 0) {
      throw new RegistryError("REGISTRY_INVALID_REPO",
        `repo ${id}: allowedAgents must be a non-empty array`);
    }
  }
}

function validateRoles(roles) {
  const required = [
    "orchestrator", "planner", "coder", "restricted-coder",
    "reviewer", "tester", "documenter", "security_reviewer"
  ];
  for (const r of required) {
    if (!roles[r]) {
      throw new RegistryError("REGISTRY_MISSING_ROLE", `missing role ${r}`);
    }
  }
  const o = roles.orchestrator.denyActions || [];
  for (const must of ["code.write", "artifact.get.raw_restricted", "approval.respond"]) {
    if (!o.includes(must)) {
      throw new RegistryError("REGISTRY_INVARIANT",
        `orchestrator must deny '${must}'`);
    }
  }
}

function validateCross(reg) {
  for (const [repoId, repo] of Object.entries(reg.repositories)) {
    if (repo.classification !== "restricted") continue;
    for (const agentId of repo.allowedAgents) {
      const agent = reg.agents[agentId];
      if (!agent) {
        throw new RegistryError("REGISTRY_INVARIANT",
          `repo ${repoId} allows unknown agent ${agentId}`);
      }
      if (!agent.allowedClassifications.includes("restricted")) {
        throw new RegistryError("REGISTRY_INVARIANT",
          `repo ${repoId} (restricted) allows agent ${agentId} that lacks 'restricted'`);
      }
    }
  }
}

export function loadRegistries({ policiesDir }) {
  const agentsFile = path.join(policiesDir, "agent-capabilities.json");
  const reposFile  = path.join(policiesDir, "repositories.json");
  const rolesFile  = path.join(policiesDir, "roles.json");

  const a = readJson(agentsFile, "agent-capabilities.json");
  const r = readJson(reposFile, "repositories.json");
  const ro = readJson(rolesFile, "roles.json");

  validateAgents(a.agents);
  validateRepos(r.repositories);
  validateRoles(ro.roles);

  const reg = {
    agents: a.agents,
    repositories: r.repositories,
    roles: ro.roles,
    protectedBranches: a.protectedBranches || []
  };

  validateCross(reg);

  return {
    getAgent(id)     { return reg.agents[id] ?? null; },
    getRepo(id)      { return reg.repositories[id] ?? null; },
    getRole(id)      { return reg.roles[id] ?? null; },
    getProtectedBranches() { return [...reg.protectedBranches]; },
    raw() { return JSON.parse(JSON.stringify(reg)); }
  };
}
```

### 3. Test

`tests/gateway/registry_loader.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import path from "node:path";
import fs from "node:fs";
import os from "node:os";

import { loadRegistries, RegistryError } from "../../gateway/src/core/registry.js";

const REAL = path.resolve("policies");

test("loads_all_registries", () => {
  const reg = loadRegistries({ policiesDir: REAL });
  assert.ok(reg.getAgent("gemini-cli"));
});

test("get_agent_returns_agent_config", () => {
  const reg = loadRegistries({ policiesDir: REAL });
  assert.ok(reg.getAgent("claude-code").allowedRoles.includes("orchestrator"));
});

test("get_repo_returns_repo_config", () => {
  const reg = loadRegistries({ policiesDir: REAL });
  assert.equal(reg.getRepo("cvision").classification, "restricted");
});

test("get_role_returns_role_config", () => {
  const reg = loadRegistries({ policiesDir: REAL });
  assert.ok(reg.getRole("orchestrator").denyActions.includes("code.write"));
});

test("invalid_registry_fails_fast", () => {
  const tmp = fs.mkdtempSync(path.join(os.tmpdir(), "reg-"));
  fs.writeFileSync(path.join(tmp, "agent-capabilities.json"), "{not json");
  fs.writeFileSync(path.join(tmp, "repositories.json"), "{}");
  fs.writeFileSync(path.join(tmp, "roles.json"), "{}");
  assert.throws(() => loadRegistries({ policiesDir: tmp }), RegistryError);
});

test("missing_registry_fails_fast", () => {
  const tmp = fs.mkdtempSync(path.join(os.tmpdir(), "reg-"));
  assert.throws(() => loadRegistries({ policiesDir: tmp }), /REGISTRY_MISSING/);
});

test("cross_invariant_restricted_repo_with_non_restricted_agent_fails", () => {
  const tmp = fs.mkdtempSync(path.join(os.tmpdir(), "reg-"));
  fs.writeFileSync(path.join(tmp, "agent-capabilities.json"), JSON.stringify({
    version: 1,
    agents: {
      "claude-code": { allowedClassifications: ["unrestricted"], allowedRoles: ["coder"] }
    }
  }));
  fs.writeFileSync(path.join(tmp, "repositories.json"), JSON.stringify({
    version: 1,
    repositories: {
      cvision: { classification: "restricted", allowedAgents: ["claude-code"] }
    }
  }));
  fs.writeFileSync(path.join(tmp, "roles.json"), fs.readFileSync(path.join(REAL, "roles.json")));
  assert.throws(() => loadRegistries({ policiesDir: tmp }), /REGISTRY_INVARIANT/);
});
```
