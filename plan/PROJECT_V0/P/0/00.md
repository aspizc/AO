### 2. Adapter

```javascript
import { BaseAdapter, assertSafeCwd } from "./base_adapter.js";
import { append as auditAppend } from "../core/audit.js";

export class CodexAdapter extends BaseAdapter {
  constructor(opts) { super({ id: "codex", ...opts }); }

  _checkEnabled() {
    const enabled = this.registries.getAgent("codex")?.enabled ?? false;
    if (!enabled) {
      const err = new Error("codex adapter is disabled in registry (B/0/0). Enable + set AGENTS_CODEX_BIN to use.");
      err.code = "ADAPTER_DISABLED";
      throw err;
    }
  }

  async delegate({ cwd, prompt, traceId, role, repo }) {
    this._checkEnabled();
    const safeCwd = assertSafeCwd(cwd, this.config.repoRoots);
    auditAppend({ type: "SESSION_STARTED", traceId, agent: "codex", role, mode: "headless" });
    if (this.config.dryRun) {
      const r = { stdout: `[dry-run codex]\n${prompt}`, stderr: "", exitCode: 0, dryRun: true };
      auditAppend({ type: "SESSION_CLOSED", traceId, agent: "codex", role, exitCode: 0, mode: "headless" });
      return r;
    }
    throw new Error("codex real mode not implemented in MVP");
  }

  async spawn() { this._checkEnabled(); throw new Error("codex spawn not implemented in MVP"); }
  async ask()   { this._checkEnabled(); throw new Error("codex ask not implemented in MVP"); }
  async view()  { this._checkEnabled(); throw new Error("codex view not implemented in MVP"); }
  async kill()  { this._checkEnabled(); throw new Error("codex kill not implemented in MVP"); }
}
```

Anadir `docs/adapters/codex.md`:

```markdown
# Codex adapter (optional, MVP-disabled)

Codex is registered as an adapter but **disabled** by default. To enable:

1. Set `enabled: true` in `policies/agent-capabilities.json` for `codex`.
2. Set `AGENTS_CODEX_BIN` to the binary path.
3. Run dry-run: `AGENTS_DRY_RUN=1` and confirm tests pass.
4. Real mode requires implementing the headless and tmux paths (post-MVP).

This adapter is **not** allowed for `restricted` repositories
(enforced by registry + policy).
```

### 4. Test

`tests/gateway/codex_adapter.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import { CodexAdapter } from "../../gateway/src/adapters/codex_adapter.js";

const fakeRegistries = (enabled) => ({
  getAgent: () => ({ enabled, allowedClassifications: ["unrestricted","internal"] }),
  getRepo: () => null, getRole: () => ({}), getProtectedBranches: () => []
});

test("disabled_codex_returns_clear_error", async () => {
  const a = new CodexAdapter({ config: { dryRun: true, repoRoots: ["/tmp"] }, registries: fakeRegistries(false) });
  await assert.rejects(() => a.delegate({ cwd: "/tmp", prompt: "x", traceId: "t", role: "coder" }),
                       /disabled/);
});

test("enabled_dry_run_delegate_works", async () => {
  const a = new CodexAdapter({ config: { dryRun: true, repoRoots: ["/tmp"] }, registries: fakeRegistries(true) });
  const r = await a.delegate({ cwd: "/tmp", prompt: "x", traceId: "t", role: "coder" });
  assert.equal(r.exitCode, 0);
});
```
