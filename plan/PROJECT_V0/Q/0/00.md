## ¿Por que existe?

Antes de pensar async, hay que pensar **estado**. Un approval tiene un ciclo de vida claro: `pending → granted | denied | expired`. Cualquier transicion fuera de ese grafo es un bug (TM-07: replay, double grant). Anclamos las invariantes en el repo + tests.

## ¿Que hay que hacer?

Extender `approval_repo.js` con:

- `createPending(row)` — inserta como `pending`.
- `getById(approvalId)`.
- `decide(approvalId, status, decidedBy, note)` con guard: solo si current `pending`.

### 2. Repo

```javascript
import { getDb } from "../state.js";

const VALID_TRANSITIONS = {
  pending: new Set(["granted", "denied", "expired"]),
};

export class ApprovalStateError extends Error {
  constructor(code, msg) { super(msg); this.code = code; }
}

export function createPendingApproval(row) {
  getDb().prepare(`
    INSERT INTO approvals(approval_id, trace_id, requested_by, action, status, created_at, payload)
    VALUES (@approvalId, @traceId, @requestedBy, @action, 'pending', @createdAt, @payload)
  `).run(row);
  return row;
}

export function getApproval(approvalId) {
  return getDb().prepare("SELECT * FROM approvals WHERE approval_id = ?").get(approvalId) || null;
}

export function decideApproval(approvalId, status, decidedBy, note) {
  if (!VALID_TRANSITIONS.pending.has(status)) {
    throw new ApprovalStateError("INVALID_STATUS", `invalid target status ${status}`);
  }
  const row = getApproval(approvalId);
  if (!row) throw new ApprovalStateError("NOT_FOUND", `approval ${approvalId} not found`);
  if (row.status !== "pending") {
    throw new ApprovalStateError("ALREADY_DECIDED", `approval ${approvalId} is ${row.status}`);
  }
  const decidedAt = new Date().toISOString();
  getDb().prepare(`
    UPDATE approvals SET status = ?, decided_at = ?, decided_by = ?,
           payload = COALESCE(?, payload)
    WHERE approval_id = ? AND status = 'pending'
  `).run(status, decidedAt, decidedBy || null, note || null, approvalId);
  return { ...row, status, decided_at: decidedAt, decided_by: decidedBy || null };
}

export function listPending(traceId) {
  return getDb().prepare("SELECT * FROM approvals WHERE status = 'pending' AND (trace_id = ? OR ? IS NULL)")
                .all(traceId || null, traceId || null);
}
```

## Errores comunes

- ❌ Permitir UPDATE sin `WHERE status = 'pending'`. Cierra TM-07.
- ❌ Tener metodo separado `grant` / `deny` con codigo duplicado. Usa `decideApproval(status)`.
