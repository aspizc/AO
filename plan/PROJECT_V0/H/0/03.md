## ¿Por que existe?

Los services no deben hacer `if (agent === 'gemini-cli') { gemini.spawn() }`. En su lugar piden al registry: `getAdapter(agentId).spawn(args)`. Asi K (agent service) no se acopla a I/O/P.

## ¿Que hay que hacer?

`gateway/src/adapters/index.js` con `createAdapterRegistry({ config, registries })`. Permite registrar adapters reales (Stages I/O/P) o inyectar mocks en tests.

### 2. `gateway/src/adapters/index.js`

```javascript
export class UnknownAgentError extends Error {
  constructor(agentId) { super(`unknown agent ${agentId}`); this.agentId = agentId; }
}

export function createAdapterRegistry({ config, registries }) {
  const adapters = new Map();

  return {
    register(agentId, adapter) {
      if (adapters.has(agentId)) {
        throw new Error(`adapter already registered for ${agentId}`);
      }
      adapters.set(agentId, adapter);
    },
    get(agentId) {
      const a = adapters.get(agentId);
      if (!a) throw new UnknownAgentError(agentId);
      return a;
    },
    has(agentId) { return adapters.has(agentId); },
    list() { return [...adapters.keys()]; },
    config, registries
  };
}
```

### 3. Test

`tests/gateway/adapter_registry.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import { createAdapterRegistry, UnknownAgentError } from "../../gateway/src/adapters/index.js";

test("unknown_agent_returns_clear_error", () => {
  const reg = createAdapterRegistry({ config: {}, registries: {} });
  assert.throws(() => reg.get("nope"), UnknownAgentError);
});

test("registered_adapter_can_be_retrieved", () => {
  const reg = createAdapterRegistry({ config: {}, registries: {} });
  const fake = { id: "x", spawn: async () => "ok" };
  reg.register("x", fake);
  assert.equal(reg.get("x"), fake);
});

test("duplicate_registration_fails", () => {
  const reg = createAdapterRegistry({ config: {}, registries: {} });
  reg.register("x", {});
  assert.throws(() => reg.register("x", {}), /already registered/);
});
```
