## ¿Por que existe?

Los nombres de sesion tmux se usan en comandos shell. Un `traceId` con caracteres raros, o un agente con `/`, rompe `tmux send-keys` y abre command injection. Centralizar la construccion del target evita errores y permite cambiar el formato (ej. `ag-` prefix por config).

## ¿Que hay que hacer?

Crear `gateway/src/adapters/session_naming.js` con `buildTmuxTarget({ traceId, agent, role, prefix })`. Sanitiza inputs.

### 2. `gateway/src/adapters/session_naming.js`

```javascript
function safeChunk(s) {
  return String(s ?? "").toLowerCase().replace(/[^a-z0-9-]+/g, "-").slice(0, 24).replace(/^-|-$/g, "");
}

function displayAgent(agent) {
  return safeChunk(String(agent || "").replace(/-cli$/, ""));
}

export function buildTmuxTarget({ traceId, agent, role, prefix = "ag-" }) {
  const tr = safeChunk(traceId);
  const ag = displayAgent(agent);
  const ro = safeChunk(role);
  if (!tr || !ag || !ro) {
    throw new TypeError("buildTmuxTarget requires traceId, agent and role");
  }
  return `${prefix}${tr}-${ag}-${ro}`.slice(0, 96);
}
```

### 3. Test

`tests/gateway/session_naming.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import { buildTmuxTarget } from "../../gateway/src/adapters/session_naming.js";

test("builds_expected_tmux_target", () => {
  const t = buildTmuxTarget({ traceId: "tr-abc123", agent: "gemini-cli", role: "restricted-coder" });
  assert.equal(t, "ag-tr-abc123-gemini-restricted-coder");
});

test("removes_cli_suffix_from_agent_for_display", () => {
  const t = buildTmuxTarget({ traceId: "x", agent: "claude-code", role: "coder" });
  assert.ok(t.includes("-claude-code-"));
  const u = buildTmuxTarget({ traceId: "x", agent: "gemini-cli", role: "coder" });
  assert.ok(u.includes("-gemini-coder"));
});

test("sanitizes_invalid_chars", () => {
  const t = buildTmuxTarget({ traceId: "tr/../x; rm -rf /", agent: "g", role: "c" });
  assert.match(t, /^[a-z0-9-]+$/);
});

test("throws_when_required_missing", () => {
  assert.throws(() => buildTmuxTarget({ traceId: "", agent: "g", role: "c" }), TypeError);
});
```
