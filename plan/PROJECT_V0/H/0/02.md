## ¿Por que existe?

Cierra TM-04 (cwd bypass). Aunque la policy diga "allow", si un adapter recibe `cwd` apuntando fuera del workspace (path tradicional, `..`, symlink), todo el sistema queda comprometido. El guard se aplica **antes** del policy y antes del spawn.

Tambien define el contrato comun de adapters: `delegate/spawn/ask/view/kill`. Sin contrato, cada adapter es un mundo y los services no pueden tratarlos uniformemente.

## ¿Que hay que hacer?

1. `gateway/src/adapters/base_adapter.js` con clase `BaseAdapter` (interfaz declarativa) y `assertSafeCwd`.
2. `realpath` antes de comparar contra allowlist.
3. Tests adversariales: `..`, symlink fuera, path absoluto fuera.

### 2. `gateway/src/adapters/base_adapter.js`

```javascript
import fs from "node:fs";
import path from "node:path";

export class CwdViolation extends Error {
  constructor(msg, details) { super(msg); this.name = "CwdViolation"; this.details = details; }
}

export function assertSafeCwd(cwd, allowedRoots) {
  if (!cwd) throw new CwdViolation("missing cwd");
  if (!Array.isArray(allowedRoots) || allowedRoots.length === 0) {
    throw new CwdViolation("AGENTS_REPO_ROOTS not configured");
  }
  let resolved;
  try {
    resolved = fs.realpathSync(cwd);
  } catch (err) {
    throw new CwdViolation(`cwd does not exist or is unreachable: ${cwd}`, { err: String(err) });
  }
  const ok = allowedRoots.some((root) => {
    const r = fs.realpathSync(root);
    return resolved === r || resolved.startsWith(r + path.sep);
  });
  if (!ok) {
    throw new CwdViolation(`cwd ${resolved} is outside allowed roots`, { allowedRoots });
  }
  return resolved;
}

export class BaseAdapter {
  /** @param {{ id: string, config: object, registries: object }} _ */
  constructor({ id, config, registries }) {
    this.id = id; this.config = config; this.registries = registries;
  }
  async delegate(_args) { throw new Error(`adapter ${this.id} must implement delegate`); }
  async spawn(_args)    { throw new Error(`adapter ${this.id} must implement spawn`); }
  async ask(_args)      { throw new Error(`adapter ${this.id} must implement ask`); }
  async view(_args)     { throw new Error(`adapter ${this.id} must implement view`); }
  async kill(_args)     { throw new Error(`adapter ${this.id} must implement kill`); }
}
```

### 3. Test

`tests/gateway/base_adapter.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import fs from "node:fs";
import os from "node:os";
import path from "node:path";
import { assertSafeCwd, CwdViolation } from "../../gateway/src/adapters/base_adapter.js";

function tmpRoot() {
  const r = fs.mkdtempSync(path.join(os.tmpdir(), "root-"));
  return fs.realpathSync(r);
}

test("cwd_inside_allowed_root_passes", () => {
  const root = tmpRoot();
  const inner = path.join(root, "inner");
  fs.mkdirSync(inner);
  assert.equal(assertSafeCwd(inner, [root]), inner);
});

test("cwd_outside_allowed_root_fails", () => {
  const root = tmpRoot();
  const outside = tmpRoot();
  assert.throws(() => assertSafeCwd(outside, [root]), CwdViolation);
});

test("realpath_escape_via_symlink_is_rejected", () => {
  const root = tmpRoot();
  const outside = tmpRoot();
  const linkInside = path.join(root, "trap");
  fs.symlinkSync(outside, linkInside);
  assert.throws(() => assertSafeCwd(linkInside, [root]), CwdViolation);
});

test("missing_cwd_or_roots_fails", () => {
  assert.throws(() => assertSafeCwd("", ["/tmp"]), CwdViolation);
  assert.throws(() => assertSafeCwd("/tmp", []), CwdViolation);
});

test("dot_dot_path_normalized_and_rejected", () => {
  const root = tmpRoot();
  assert.throws(() => assertSafeCwd(path.join(root, "..", "..", "/tmp"), [root]), CwdViolation);
});
```

## Errores comunes a evitar

- ❌ Comparar `cwd` sin `realpath` primero. Symlinks bypass.
- ❌ Comparar `startsWith` sin separador. `/repo/foo` matchea `/repo/foo-evil`.
- ❌ Permitir un default global "all paths" si no hay allowlist. Default-deny.
