### 2. `gateway/src/adapters/tmux_client.js`

Portar (no copiar literal: limpia, anade JSDoc, retira partes Gemini-specific):

```javascript
import { spawn, spawnSync } from "node:child_process";

export function isTmuxAvailable() {
  try {
    return spawnSync("tmux", ["-V"]).status === 0;
  } catch { return false; }
}

export function buildNewSessionCmd({ target, cwd }) {
  return ["new-session", "-d", "-s", target, "-c", cwd];
}

export function buildSendKeysCmd({ target, line }) {
  return ["send-keys", "-t", target, line, "Enter"];
}

export function buildCapturePaneCmd({ target, lines = 200 }) {
  return ["capture-pane", "-pt", target, "-S", `-${lines}`];
}

export function buildKillSessionCmd({ target }) {
  return ["kill-session", "-t", target];
}

export function tmuxSync(args, opts = {}) {
  return spawnSync("tmux", args, { encoding: "utf-8", ...opts });
}

export function tmuxAsync(args, opts = {}) {
  return spawn("tmux", args, opts);
}
```

### 3. Test

`tests/gateway/tmux_client.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import {
  isTmuxAvailable, buildNewSessionCmd, buildSendKeysCmd,
  buildCapturePaneCmd, buildKillSessionCmd, tmuxSync
} from "../../gateway/src/adapters/tmux_client.js";

test("builds_tmux_commands", () => {
  assert.deepEqual(
    buildNewSessionCmd({ target: "ag-x", cwd: "/tmp" }),
    ["new-session", "-d", "-s", "ag-x", "-c", "/tmp"]
  );
  assert.deepEqual(
    buildSendKeysCmd({ target: "ag-x", line: "ls" }),
    ["send-keys", "-t", "ag-x", "ls", "Enter"]
  );
  assert.deepEqual(
    buildCapturePaneCmd({ target: "ag-x", lines: 50 }),
    ["capture-pane", "-pt", "ag-x", "-S", "-50"]
  );
});

test("create_and_kill_session_if_tmux_available", () => {
  if (!isTmuxAvailable()) return;
  const target = `agtest-${Date.now()}`;
  const r1 = tmuxSync(buildNewSessionCmd({ target, cwd: "/tmp" }));
  assert.equal(r1.status, 0);
  const r2 = tmuxSync(buildKillSessionCmd({ target }));
  assert.equal(r2.status, 0);
});
```
