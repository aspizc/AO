## ¿Por que existe?

ADR-003 dice: **policy antes del spawn**. Cierra TM-04 a nivel de adapter (defense in depth, ademas de la tool MCP). Tambien aqui se afirma que cualquier excepcion del adapter genera un evento `ERROR` en audit, no solo un crash silencioso.

## ¿Que hay que hacer?

Anadir un preflight a `delegate` y `spawn` que invoque al policy engine antes de cualquier `spawnSync`. Audit `ERROR` en catch.

### 2. Refactor `delegate/spawn`

Cabecera comun:

```javascript
import { evaluate } from "../core/policy_engine.js";

function preflight({ registries }, ctx) {
  const r = evaluate(ctx, registries);
  if (r.decision !== "allow") {
    const err = new Error(`policy denied: ${r.reason}`);
    err.policy = r;
    throw err;
  }
  return r;
}

// Inside delegate:
async delegate({ cwd, prompt, traceId, role, repo }) {
  try {
    preflight(this, {
      agent: "gemini-cli", role, action: "agent.delegate", repo: repo || null
    });
    // ... existing spawn logic
  } catch (err) {
    auditAppend({
      type: "ERROR", traceId, agent: "gemini-cli", role,
      where: "delegate", error: err.message, policy: err.policy || null
    });
    throw err;
  }
}
```

### 3. Test

`tests/gateway/gemini_policy_audit.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import fs from "node:fs"; import os from "node:os"; import path from "node:path";
import { GeminiAdapter } from "../../gateway/src/adapters/gemini_adapter.js";
import { configureAudit, query, _resetForTests as resetAudit } from "../../gateway/src/core/audit.js";

function fakeRegistries({ deny = false } = {}) {
  return {
    getAgent: () => ({ allowedClassifications: ["unrestricted","internal","restricted"], allowedRoles: ["coder","restricted-coder"] }),
    getRepo: () => null,
    getRole: () => ({ allowActions: ["code.read","code.write","agent.delegate"], denyActions: deny ? ["agent.delegate"] : [] }),
    getProtectedBranches: () => []
  };
}

function setup() {
  resetAudit();
  const root = fs.realpathSync(fs.mkdtempSync(path.join(os.tmpdir(), "r-")));
  const audit = path.join(fs.mkdtempSync(path.join(os.tmpdir(), "a-")), "audit.jsonl");
  configureAudit({ auditLog: audit });
  return { root };
}

test("policy_deny_prevents_spawn", async () => {
  const { root } = setup();
  const a = new GeminiAdapter({ config: { dryRun: true, repoRoots: [root] }, registries: fakeRegistries({ deny: true }) });
  await assert.rejects(() => a.delegate({ cwd: root, prompt: "x", traceId: "td", role: "coder" }));
  const events = await query({ traceId: "td" });
  // No SESSION_STARTED expected because preflight failed
  assert.ok(!events.some((e) => e.type === "SESSION_STARTED"));
  assert.ok(events.some((e) => e.type === "ERROR"));
});

test("session_started_and_closed_audited", async () => {
  const { root } = setup();
  const a = new GeminiAdapter({ config: { dryRun: true, repoRoots: [root] }, registries: fakeRegistries() });
  await a.delegate({ cwd: root, prompt: "x", traceId: "ts", role: "coder" });
  const events = await query({ traceId: "ts" });
  assert.ok(events.some((e) => e.type === "SESSION_STARTED"));
  assert.ok(events.some((e) => e.type === "SESSION_CLOSED"));
});

test("adapter_error_audited", async () => {
  const { root } = setup();
  const a = new GeminiAdapter({ config: { dryRun: true, repoRoots: [root] }, registries: fakeRegistries() });
  await assert.rejects(() => a.delegate({ cwd: "/no/such", prompt: "x", traceId: "te", role: "coder" }));
  const events = await query({ traceId: "te" });
  assert.ok(events.some((e) => e.type === "ERROR"));
});
```
