## ¿Por que existe?

Tareas largas o interactivas no caben en delegate (run-once). Necesitamos `spawn` (crea sesion tmux), `ask` (envia prompt y espera estabilizacion), `view` (snapshot del pane), `kill` (cierra). Esto hereda el patron del experimento `gemini-orchestrator` y respeta `correcciones_orquestador_hijos_tmux.md` (tmux es **observacion + intervencion**).

## ¿Que hay que hacer?

Anadir metodos a `GeminiAdapter`: `spawn`, `ask`, `view`, `kill`. Reusa `tmux_client` y `session_naming`.

### 2. Extender adapter

```javascript
import {
  isTmuxAvailable, tmuxSync,
  buildNewSessionCmd, buildSendKeysCmd,
  buildCapturePaneCmd, buildKillSessionCmd
} from "./tmux_client.js";
import { buildTmuxTarget } from "./session_naming.js";

// inside class GeminiAdapter:

async spawn({ cwd, traceId, role }) {
  const safeCwd = assertSafeCwd(cwd, this.config.repoRoots);
  const target = buildTmuxTarget({
    traceId, agent: "gemini-cli", role,
    prefix: this.config.tmuxPrefix
  });
  auditAppend({ type: "SESSION_STARTED", traceId, agent: "gemini-cli", role, mode: "supervised", tmuxTarget: target });

  if (!this.config.dryRun) {
    if (!isTmuxAvailable()) throw new Error("tmux is required for supervised mode");
    const r = tmuxSync(buildNewSessionCmd({ target, cwd: safeCwd }));
    if (r.status !== 0) throw new Error(`tmux new-session failed: ${r.stderr}`);
    // start gemini inside tmux
    tmuxSync(buildSendKeysCmd({ target, line: this.config.geminiBin || "gemini" }));
  }

  return {
    sessionId: target,
    tmuxTarget: target,
    attachCommand: `tmux attach -t ${target}`,
    dryRun: !!this.config.dryRun
  };
}

async ask({ tmuxTarget, prompt, traceId, role }) {
  if (this.config.dryRun) {
    return { snapshot: `[dry-run ask]\n${prompt}\n[ok]`, dryRun: true };
  }
  tmuxSync(buildSendKeysCmd({ target: tmuxTarget, line: prompt }));
  // simple wait + capture (real impl would stabilize loops; keep MVP simple)
  await new Promise((r) => setTimeout(r, 1500));
  const cap = tmuxSync(buildCapturePaneCmd({ target: tmuxTarget, lines: 400 }));
  auditAppend({ type: "SESSION_INPUT", traceId, agent: "gemini-cli", role, tmuxTarget, prompt: prompt.slice(0, 200) });
  return { snapshot: cap.stdout || "", dryRun: false };
}

async view({ tmuxTarget }) {
  if (this.config.dryRun) return { snapshot: "[dry-run view]", dryRun: true };
  const cap = tmuxSync(buildCapturePaneCmd({ target: tmuxTarget, lines: 400 }));
  return { snapshot: cap.stdout || "", dryRun: false };
}

async kill({ tmuxTarget, traceId, role }) {
  if (!this.config.dryRun) tmuxSync(buildKillSessionCmd({ target: tmuxTarget }));
  auditAppend({ type: "SESSION_CLOSED", traceId, agent: "gemini-cli", role, tmuxTarget, mode: "supervised" });
  return { closed: true, dryRun: !!this.config.dryRun };
}
```

### 3. Test

`tests/gateway/gemini_supervised.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import fs from "node:fs"; import os from "node:os"; import path from "node:path";
import { GeminiAdapter } from "../../gateway/src/adapters/gemini_adapter.js";
import { configureAudit, _resetForTests as resetAudit } from "../../gateway/src/core/audit.js";

function setup() {
  resetAudit();
  const root = fs.realpathSync(fs.mkdtempSync(path.join(os.tmpdir(), "r-")));
  configureAudit({ auditLog: path.join(fs.mkdtempSync(path.join(os.tmpdir(), "a-")), "audit.jsonl") });
  return { root };
}

test("dry_run_spawn_returns_session_info", async () => {
  const { root } = setup();
  const a = new GeminiAdapter({ config: { dryRun: true, repoRoots: [root], tmuxPrefix: "ag-" }, registries: {} });
  const s = await a.spawn({ cwd: root, traceId: "tr1", role: "coder" });
  assert.match(s.tmuxTarget, /^ag-/);
  assert.match(s.attachCommand, /^tmux attach/);
});

test("dry_run_ask_returns_response", async () => {
  const { root } = setup();
  const a = new GeminiAdapter({ config: { dryRun: true, repoRoots: [root], tmuxPrefix: "ag-" }, registries: {} });
  const r = await a.ask({ tmuxTarget: "ag-x", prompt: "hi", traceId: "t", role: "c" });
  assert.ok(r.dryRun);
});

test("view_returns_snapshot_in_dry_run", async () => {
  const { root } = setup();
  const a = new GeminiAdapter({ config: { dryRun: true, repoRoots: [root] }, registries: {} });
  const r = await a.view({ tmuxTarget: "ag-x" });
  assert.ok(typeof r.snapshot === "string");
});

test("kill_marks_session_closed_in_audit", async () => {
  const { root } = setup();
  const a = new GeminiAdapter({ config: { dryRun: true, repoRoots: [root] }, registries: {} });
  await a.kill({ tmuxTarget: "ag-x", traceId: "tk", role: "c" });
});
```
