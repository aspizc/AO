## ¿Por que existe?

`gemini -p --yolo` es la baseline para Gemini headless del experimento. La portamos como `delegate(args)`: input → cli → captura stdout/stderr/exit. Es la forma rapida de ejecutar tareas pequenas sin sesion persistente.

`AGENTS_DRY_RUN=1` hace que CI no intente lanzar gemini real (no esta instalado en CI), pero los tests siguen verificando paths y comandos construidos.

## ¿Que hay que hacer?

`gateway/src/adapters/gemini_adapter.js` con `delegate({ cwd, prompt, traceId, role })`. Construye el comando, valida cwd, audita `SESSION_STARTED/CLOSED`, devuelve `{ stdout, stderr, exitCode }`. En dry-run devuelve mock determinista.

### 2. Adapter

```javascript
import { spawnSync } from "node:child_process";
import { BaseAdapter, assertSafeCwd } from "./base_adapter.js";
import { append as auditAppend } from "../core/audit.js";

export class GeminiAdapter extends BaseAdapter {
  constructor(opts) {
    super({ id: "gemini-cli", ...opts });
  }

  async delegate({ cwd, prompt, traceId, role }) {
    const safeCwd = assertSafeCwd(cwd, this.config.repoRoots);
    auditAppend({
      type: "SESSION_STARTED", traceId, agent: "gemini-cli", role, mode: "headless"
    });

    if (this.config.dryRun) {
      const r = { stdout: `[dry-run]\nprompt=${prompt}\ncwd=${safeCwd}`, stderr: "", exitCode: 0, dryRun: true };
      auditAppend({ type: "SESSION_CLOSED", traceId, agent: "gemini-cli", role, exitCode: 0, mode: "headless" });
      return r;
    }

    const bin = this.config.geminiBin || "gemini";
    const proc = spawnSync(bin, ["-p", "--yolo", prompt], {
      cwd: safeCwd, encoding: "utf-8", timeout: this.config.adapterTimeoutMs || 600_000
    });
    const r = { stdout: proc.stdout || "", stderr: proc.stderr || "", exitCode: proc.status ?? -1, dryRun: false };
    auditAppend({ type: "SESSION_CLOSED", traceId, agent: "gemini-cli", role, exitCode: r.exitCode, mode: "headless" });
    return r;
  }
}
```

### 3. Test

`tests/gateway/gemini_delegate.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import fs from "node:fs"; import os from "node:os"; import path from "node:path";
import { GeminiAdapter } from "../../gateway/src/adapters/gemini_adapter.js";
import { configureAudit, query, _resetForTests as resetAudit } from "../../gateway/src/core/audit.js";

function setup() {
  resetAudit();
  const root = fs.realpathSync(fs.mkdtempSync(path.join(os.tmpdir(), "r-")));
  const audit = path.join(fs.mkdtempSync(path.join(os.tmpdir(), "a-")), "audit.jsonl");
  configureAudit({ auditLog: audit });
  return { root, audit };
}

test("dry_run_delegate_returns_mock", async () => {
  const { root } = setup();
  const a = new GeminiAdapter({
    config: { dryRun: true, repoRoots: [root] }, registries: {}
  });
  const r = await a.delegate({ cwd: root, prompt: "hi", traceId: "t1", role: "coder" });
  assert.equal(r.exitCode, 0);
  assert.ok(r.dryRun);
});

test("delegate_audits_session_lifecycle", async () => {
  const { root } = setup();
  const a = new GeminiAdapter({ config: { dryRun: true, repoRoots: [root] }, registries: {} });
  await a.delegate({ cwd: root, prompt: "hi", traceId: "t-aud", role: "coder" });
  const events = await query({ traceId: "t-aud" });
  assert.ok(events.some((e) => e.type === "SESSION_STARTED"));
  assert.ok(events.some((e) => e.type === "SESSION_CLOSED"));
});

test("cwd_outside_allowlist_throws", async () => {
  const { root } = setup();
  const a = new GeminiAdapter({ config: { dryRun: true, repoRoots: [root] }, registries: {} });
  const outside = fs.realpathSync(fs.mkdtempSync(path.join(os.tmpdir(), "out-")));
  await assert.rejects(() => a.delegate({ cwd: outside, prompt: "x", traceId: "t", role: "c" }));
});
```

## Errores comunes a evitar

- ❌ Saltarse `assertSafeCwd` en dry-run. El guard es independiente de la ejecucion real.
- ❌ Pasar `prompt` con shell injection. `spawnSync` con array es seguro; no uses `exec`.
- ❌ Olvidar el timeout. Subprocess sin timeout = potencial DoS.
