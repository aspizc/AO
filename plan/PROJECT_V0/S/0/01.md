### 2. Tool

```javascript
import { defineTool, z } from "./tool_helpers.js";
import { newMessageId } from "../core/ids.js";
import * as repo from "../core/repositories/message_repo.js";
import { append as auditAppend } from "../core/audit.js";

export function buildMessageTools() {
  return [
    defineTool({
      name: "message.send",
      description: "Send a message between participants of an orchestration trace.",
      schema: z.object({
        traceId: z.string(), fromId: z.string(), toId: z.string(),
        body: z.string()
      }),
      handler: async (a) => {
        const row = {
          messageId: newMessageId(), traceId: a.traceId,
          fromId: a.fromId, toId: a.toId, body: a.body,
          createdAt: new Date().toISOString()
        };
        repo.createMessage(row);
        auditAppend({ type: "MESSAGE_SENT", traceId: a.traceId, messageId: row.messageId,
                      fromId: a.fromId, toId: a.toId });
        return row;
      }
    }),
    defineTool({
      name: "message.list",
      description: "List all messages in a traceId.",
      schema: z.object({ traceId: z.string() }),
      handler: async ({ traceId }) => repo.listMessagesByTrace(traceId)
    }),
    defineTool({
      name: "message.reply",
      description: "Reply to a previous message; result is a new message scoped to same trace.",
      schema: z.object({
        traceId: z.string(), parentMessageId: z.string(),
        fromId: z.string(), toId: z.string(), body: z.string()
      }),
      handler: async (a) => {
        const parent = repo.getMessageScopedToTrace(a.parentMessageId, a.traceId);
        if (!parent) return { error: "PARENT_NOT_FOUND" };
        return await this.handler({ ...a }); // wire to message.send via service or duplicate row insert
      }
    })
  ];
}
```

> Nota: `reply` puede simplemente delegar a `send` con un campo `parentMessageId` opcional en el body, o usar un repo extendido. Mantenlo simple en MVP.
