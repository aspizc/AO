## ¿Por que existe?

Si cada tool MCP escribe SQL inline, en 3 stages el codigo es ingobernable y los bypass se multiplican (¿quien valida un INSERT en `tasks`? ¿cuando se aplica policy?). Repository pattern = un solo lugar por tabla, un solo conjunto de prepared statements, un solo punto donde anclar transacciones.

## ¿Que hay que hacer?

Crear `gateway/src/core/repositories/` con un modulo por tabla, exportando funciones puras: `create`, `getById`, `listByTraceId`, etc. Sin acoplar a MCP.

### 2. Estructura

Crea uno por tabla. Patron generico (ejemplo `orchestration_repo.js`):

```javascript
import { getDb } from "../state.js";

export function createOrchestration(row) {
  getDb().prepare(`
    INSERT INTO orchestration_sessions
    (session_id, trace_id, caller_agent, caller_role, status, goal, created_at)
    VALUES (@sessionId, @traceId, @callerAgent, @callerRole, @status, @goal, @createdAt)
  `).run(row);
  return row;
}

export function getOrchestrationByTraceId(traceId) {
  return getDb().prepare(
    "SELECT * FROM orchestration_sessions WHERE trace_id = ?"
  ).get(traceId) || null;
}

export function setStatus(sessionId, status) {
  getDb().prepare("UPDATE orchestration_sessions SET status=? WHERE session_id=?")
        .run(status, sessionId);
}
```

Repite el patron para `task_repo`, `session_repo`, `artifact_repo`, `policy_decision_repo`, `approval_repo`, `message_repo`. Mantenlos minimos: solo lo que el codigo de stages siguientes va a necesitar.

### 3. Test

`tests/gateway/domain_repositories.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import fs from "node:fs";
import os from "node:os";
import path from "node:path";
import { initState, _resetForTests } from "../../gateway/src/core/state.js";
import * as orch from "../../gateway/src/core/repositories/orchestration_repo.js";
import * as tasks from "../../gateway/src/core/repositories/task_repo.js";
import * as art from "../../gateway/src/core/repositories/artifact_repo.js";

function fresh() {
  _resetForTests();
  initState({ stateDb: path.join(fs.mkdtempSync(path.join(os.tmpdir(), "r-")), "x.db") });
}

test("create_and_get_orchestration", () => {
  fresh();
  orch.createOrchestration({
    sessionId: "s1", traceId: "t1", callerAgent: "claude-code",
    callerRole: "orchestrator", status: "active", goal: null,
    createdAt: "2026-01-01T00:00:00Z"
  });
  const got = orch.getOrchestrationByTraceId("t1");
  assert.equal(got.session_id, "s1");
});

test("create_task_under_trace", () => {
  fresh();
  orch.createOrchestration({sessionId:"s1", traceId:"t1", callerAgent:"x",
    callerRole:"orchestrator", status:"active", goal:null,
    createdAt:"2026-01-01T00:00:00Z"});
  tasks.createTask({taskId:"k1", traceId:"t1", assignedAgent:"gemini-cli",
    assignedRole:"restricted-coder", repo:"cvision", status:"pending",
    createdAt:"2026-01-01T00:00:00Z", closedAt:null});
  assert.equal(tasks.listTasksByTrace("t1").length, 1);
});

test("foreign_key_violation_fails", () => {
  fresh();
  assert.throws(() => tasks.createTask({taskId:"k1", traceId:"missing",
    assignedAgent:"x", assignedRole:"x", repo:null, status:"pending",
    createdAt:"now", closedAt:null}));
});

test("insert_policy_decision_is_append_only", () => {
  fresh();
  // No update API exposed; just verify there is no setter
  const mod = await import("../../gateway/src/core/repositories/policy_decision_repo.js");
  assert.ok(typeof mod.insertDecision === "function");
  assert.ok(typeof mod.updateDecision === "undefined");
});
```

## Errores comunes a evitar

- ❌ Exportar la conexion a tools/services. Solo getters/setters.
- ❌ Anadir UPDATE a `policy_decisions`. Append-only por design.
- ❌ Concatenar SQL con strings. Siempre prepared statements.
- ❌ Hacer transacciones manualmente con BEGIN/COMMIT. Usa `db.transaction(fn)`.
