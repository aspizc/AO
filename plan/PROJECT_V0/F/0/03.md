## ¿Por que existe?

Cualquier IDs predecibles permiten ataques de spoofing/replay (TM-07). Centralizar la generacion en un modulo, basado en `crypto.randomUUID()`, garantiza:

- Imposibles de adivinar.
- Format estable (auditoria los puede correlar).
- `traceId` con prefijo opcional para que el operador pueda leer logs.

## ¿Que hay que hacer?

Crear `gateway/src/core/ids.js` con generadores. Un test confirma unicidad y formato.

### 2. `gateway/src/core/ids.js`

```javascript
import crypto from "node:crypto";

const UUID_RE = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/;

function uuid() { return crypto.randomUUID(); }

function slugify(s) {
  return String(s).toLowerCase().replace(/[^a-z0-9]+/g, "-").slice(0, 24).replace(/^-|-$/g, "");
}

export function newTraceId({ prefix } = {}) {
  return prefix ? `tr-${slugify(prefix)}-${uuid()}` : `tr-${uuid()}`;
}

export const newOrchestrationId = () => `os-${uuid()}`;
export const newChildTaskId     = () => `ts-${uuid()}`;
export const newSessionId       = () => `ss-${uuid()}`;
export const newArtifactId      = () => `art-${uuid()}`;
export const newApprovalId      = () => `apr-${uuid()}`;
export const newMessageId       = () => `msg-${uuid()}`;
export const newPolicyDecisionId = () => `pd-${uuid()}`;

export function isUuid(value) {
  return UUID_RE.test(String(value));
}
```

### 3. Test

`tests/gateway/ids.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import {
  newTraceId, newChildTaskId, newSessionId,
  newArtifactId, newApprovalId, newMessageId, isUuid
} from "../../gateway/src/core/ids.js";

test("uuid_ids_are_unique", () => {
  const all = new Set();
  for (let i = 0; i < 1000; i++) all.add(newSessionId());
  assert.equal(all.size, 1000);
});

test("trace_id_is_unique", () => {
  const all = new Set();
  for (let i = 0; i < 1000; i++) all.add(newTraceId());
  assert.equal(all.size, 1000);
});

test("trace_id_accepts_optional_slug_prefix", () => {
  const id = newTraceId({ prefix: "Refactor module X" });
  assert.match(id, /^tr-refactor-module-x-/);
});

test("child_task_id_format", () => {
  const id = newChildTaskId();
  assert.ok(id.startsWith("ts-"));
  assert.ok(isUuid(id.slice(3)));
});
```

## Errores comunes a evitar

- ❌ Usar `Date.now()` o counters. Adivinables.
- ❌ Poner `traceId` como int. Hace queries faciles pero rompe seguridad.
- ❌ Slugificar mal: caracteres unicode pueden colarse. Limita a `[a-z0-9-]`.
