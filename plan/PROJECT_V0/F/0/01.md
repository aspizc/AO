## ¿Por que existe?

Migracion en disco no sirve si el Gateway no sabe abrir la DB con el path correcto y aplicar pendientes. Necesitamos un `initState(config)` que:

1. Crea el dir padre si falta.
2. Aplica migraciones pendientes.
3. Activa `foreign_keys`.
4. Expone `getDb()` a repositorios.

## ¿Que hay que hacer?

Crear `gateway/src/core/state.js` con `initState(config)` y `getDb()`. Tests con dirs temporales.

### 2. `gateway/src/core/state.js`

```javascript
import fs from "node:fs";
import path from "node:path";
import Database from "better-sqlite3";

let _db = null;

const MIGRATIONS_DIR = path.resolve("gateway/migrations");

function listMigrations() {
  return fs.readdirSync(MIGRATIONS_DIR)
    .filter((f) => f.endsWith(".sql"))
    .sort();
}

function applyPending(db) {
  db.exec("CREATE TABLE IF NOT EXISTS schema_migrations(id TEXT PRIMARY KEY, applied_at TEXT NOT NULL)");
  const applied = new Set(db.prepare("SELECT id FROM schema_migrations").all().map((r) => r.id));
  for (const file of listMigrations()) {
    const id = file.replace(/\.sql$/, "");
    if (applied.has(id)) continue;
    const sql = fs.readFileSync(path.join(MIGRATIONS_DIR, file), "utf-8");
    const tx = db.transaction(() => {
      db.exec(sql);
      db.prepare("INSERT OR IGNORE INTO schema_migrations(id, applied_at) VALUES (?, strftime('%Y-%m-%dT%H:%M:%fZ','now'))")
        .run(id);
    });
    tx();
  }
}

export function initState({ stateDb }) {
  if (!stateDb) throw new TypeError("stateDb path required");
  fs.mkdirSync(path.dirname(stateDb), { recursive: true });
  const db = new Database(stateDb);
  db.pragma("journal_mode = WAL");
  db.pragma("foreign_keys = ON");
  applyPending(db);
  _db = db;
  return db;
}

export function getDb() {
  if (!_db) throw new Error("state not initialized; call initState first");
  return _db;
}

export function _resetForTests() {
  if (_db) { _db.close(); _db = null; }
}
```

### 3. Test

`tests/gateway/state_init.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import fs from "node:fs";
import os from "node:os";
import path from "node:path";
import { initState, getDb, _resetForTests } from "../../gateway/src/core/state.js";

function tmpDb() {
  return path.join(fs.mkdtempSync(path.join(os.tmpdir(), "st-")), "state.db");
}

test("creates_db_file", () => {
  _resetForTests();
  const f = tmpDb();
  initState({ stateDb: f });
  assert.ok(fs.existsSync(f));
});

test("runs_migrations", () => {
  _resetForTests();
  initState({ stateDb: tmpDb() });
  const tables = getDb().prepare("SELECT name FROM sqlite_master WHERE type='table'")
                        .all().map((r) => r.name);
  assert.ok(tables.includes("orchestration_sessions"));
});

test("enables_foreign_keys", () => {
  _resetForTests();
  initState({ stateDb: tmpDb() });
  const r = getDb().prepare("PRAGMA foreign_keys").get();
  assert.equal(r.foreign_keys, 1);
});
```

## Errores comunes a evitar

- ❌ Llamar a `initState` mas de una vez sin reset. Cachea singleton.
- ❌ Aplicar migracion sin transaccion. Migracion incompleta = DB inconsistente.
- ❌ Olvidar `journal_mode=WAL`. Mejor concurrencia entre lectura y escritura.
