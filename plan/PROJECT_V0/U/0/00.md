## ¿Por que existe?

V4 §30 declara que el MVP esta "listo" cuando el flujo restringido completo (orchestrator humano-LLM → restricted-coder → sanitizer → reviewer → approval → push protegido) corre **sin tocar repos reales** y **sin filtrar raw**. Esta tarea cierra ese loop con un test E2E que vive en `tests/e2e/`.

A diferencia de los tests unitarios por modulo, este test orquesta el Gateway completo (con dry-run adapters) y verifica las invariantes de end-to-end: traceId compartido, audit completo, sanitizacion aplicada, raw nunca cruza el boundary.

## ¿Que hay que hacer?

Crear `tests/e2e/mvp_restricted_flow.test.js` que:

1. Arranca el Gateway in-process con `AGENTS_DRY_RUN=1`.
2. Ejecuta la secuencia canonical V4.
3. Verifica que el orchestrator nunca recibe raw.
4. Verifica audit con todos los eventos clave.

## Pre-requisitos

- Stages F-R completas (todos los servicios y tools).
- Adapters dry-run de Gemini (I) y Claude (O).
- Sanitizer (M) y artifact share con boundary check (N/0/2).

### 2. Helper de bootstrap del Gateway in-process

`tests/e2e/helpers/gateway_harness.js`:

```javascript
import { initConfig } from "../../../gateway/src/core/config.js";
import { initRegistries } from "../../../gateway/src/core/registries.js";
import { initState } from "../../../gateway/src/core/state.js";
import { init as initAudit } from "../../../gateway/src/core/audit.js";
import { mkdtempSync, rmSync } from "node:fs";
import { tmpdir } from "node:os";
import path from "node:path";

import * as orchestration from "../../../gateway/src/services/orchestration_service.js";
import * as task from "../../../gateway/src/services/task_service.js";
import * as agent from "../../../gateway/src/services/agent_service.js";
import * as artifact from "../../../gateway/src/services/artifact_service.js";
import * as approval from "../../../gateway/src/services/approval_service.js";

export function startHarness() {
  const workspace = mkdtempSync(path.join(tmpdir(), "ag-e2e-"));
  process.env.AGENTS_WORKSPACE = workspace;
  process.env.AGENTS_DRY_RUN = "1";
  process.env.AGENTS_REPO_ROOTS = path.join(workspace, "repos");

  initConfig();
  initRegistries();
  initState();
  initAudit();

  return {
    workspace,
    services: { orchestration, task, agent, artifact, approval },
    cleanup: () => rmSync(workspace, { recursive: true, force: true })
  };
}
```

### 3. Test principal

`tests/e2e/mvp_restricted_flow.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import fs from "node:fs";
import path from "node:path";
import { startHarness } from "./helpers/gateway_harness.js";

test("restricted_flow_end_to_end", async () => {
  const h = startHarness();
  try {
    const { orchestration, task, agent, artifact, approval } = h.services;
    const orch = orchestration.create({
      callerAgent: "claude-code", callerRole: "orchestrator",
      goal: "Fix bug in cvision parser"
    });

    const t = task.assign({
      traceId: orch.traceId,
      caller: { agent: "claude-code", role: "orchestrator" },
      target: { role: "restricted-coder", agent: "gemini-cli" },
      repo: "cvision",
      description: "Apply parser fix"
    });

    const session = await agent.spawn({
      agent: "gemini-cli", role: "restricted-coder", repo: "cvision",
      cwd: path.join(h.workspace, "repos/cvision"),
      traceId: orch.traceId, taskId: t.taskId
    });

    const raw = artifact.put({
      traceId: orch.traceId, taskId: t.taskId,
      kind: "raw_diff", classification: "restricted",
      producer: { agent: "gemini-cli", role: "restricted-coder" },
      bodyText: "diff with secret=AKIAFAKEKEY1234"
    });

    const sanitized = artifact.share({
      traceId: orch.traceId, artifactId: raw.artifactId,
      requester: { agent: "claude-code", role: "orchestrator" }
    });
    assert.equal(sanitized.classification, "internal");
    assert.ok(!String(sanitized.bodyText).includes("AKIAFAKEKEY"));

    const apr = approval.request({
      traceId: orch.traceId, action: "git.push.protected",
      requestedBy: "claude-code", context: { branch: "main" }
    });
    assert.equal(apr.status, "pending");
    approval.respond({
      approvalId: apr.approvalId, decision: "granted",
      respondedBy: "operator-cli", note: "release plan ok"
    });

    orchestration.complete({ traceId: orch.traceId, summary: "done" });

    const auditPath = path.join(h.workspace, "audit/events.jsonl");
    const lines = fs.readFileSync(auditPath, "utf8").trim().split("\n").map((l) => JSON.parse(l));
    const types = new Set(lines.map((e) => e.type));
    for (const ev of [
      "ORCHESTRATION_CREATED", "TASK_CREATED", "SESSION_STARTED",
      "ARTIFACT_CREATED", "ARTIFACT_SANITIZED", "ARTIFACT_SHARED",
      "APPROVAL_REQUIRED", "APPROVAL_GRANTED", "ORCHESTRATION_COMPLETED"
    ]) {
      assert.ok(types.has(ev), `missing audit event ${ev}`);
    }
  } finally {
    h.cleanup();
  }
});
```

## Errores comunes

- ❌ Devolver `raw_diff` directamente al orchestrator (TM-02). El test debe fallar si esto ocurre.
- ❌ Olvidar `cleanup` → workspace temp se acumula en disco.
