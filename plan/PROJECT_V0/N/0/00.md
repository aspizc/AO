## ¿Por que existe?

`artifact.share` es el patron explicito de **artifact-mediated communication** (V4 §17.2.2): un coder publica un artifact, el orchestrator pide compartirlo a un reviewer. El share es policy-aware: si reviewer no puede ver raw restricted, el share devuelve el sanitized linkado.

## ¿Que hay que hacer?

`gateway/src/services/artifact_share_service.js` con `share({ artifactId, requesterAgent, requesterRole, traceId })`. Aplica policy. Si necesita sanitization, busca el sanitized via `sanitized_from`.

### 2. Service

```javascript
import { evaluate } from "../core/policy_engine.js";
import * as repo from "../core/repositories/artifact_repo.js";
import { append as auditAppend } from "../core/audit.js";

export function shareArtifact({ artifactId, requesterAgent, requesterRole, traceId, registries }) {
  const row = repo.getArtifact(artifactId);
  if (!row) return { error: "NOT_FOUND" };
  if (row.trace_id !== traceId) {
    return { error: "POLICY_DENIED",
             decision: { decision: "deny", reason: "cross-trace share", ruleId: "share.cross_trace" } };
  }
  const decision = evaluate({
    agent: requesterAgent, role: requesterRole, action: "artifact.share",
    artifactKind: row.kind, artifactClassification: row.classification
  }, registries);

  let sharedArtifactId = null;
  if (decision.decision === "allow") sharedArtifactId = artifactId;
  else if (decision.decision === "allow_with_sanitization") {
    const san = repo.findSanitizedFor(artifactId);
    if (!san) return { error: "SANITIZATION_MISSING",
             decision: { decision: "deny", reason: "sanitized version missing", ruleId: "share.sanitization_missing" } };
    sharedArtifactId = san.artifact_id;
  } else {
    auditAppend({ type: "ARTIFACT_SHARE_DENIED", traceId, artifactId, requesterAgent, requesterRole, decision });
    return { error: "POLICY_DENIED", decision };
  }

  auditAppend({ type: "ARTIFACT_SHARED", traceId, sourceArtifactId: artifactId, sharedArtifactId, requesterAgent, requesterRole });
  return { decision: decision.decision, sharedArtifactId };
}
```
