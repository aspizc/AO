## ¿Por que existe?

Sin un service intermedio, las tools `agent.*` tendrian que hablar directamente con adapters. Eso rompe el patron capa: tools → services → core/adapters. Tambien hace dificil aplicar policy y persistir sesiones de forma consistente.

`AgentService` resuelve adapter desde el registry, valida policy, persiste `sessions` (F/0/2), audita lifecycle y maneja errores estructurados.

## ¿Que hay que hacer?

`gateway/src/services/agent_service.js` con `delegate/spawn/ask/view/kill`. Cada metodo: policy → adapter → persistir → audit.

### 2. Service

```javascript
import { evaluate } from "../core/policy_engine.js";
import { newSessionId } from "../core/ids.js";
import * as sess from "../core/repositories/session_repo.js";
import { append as auditAppend } from "../core/audit.js";

export class PolicyDeniedError extends Error {
  constructor(d) { super(d.reason); this.code = "POLICY_DENIED"; this.decision = d; }
}

export function createAgentService({ adapters, registries }) {
  function policy(ctx) {
    const r = evaluate(ctx, registries);
    if (r.decision !== "allow") throw new PolicyDeniedError(r);
    return r;
  }

  return {
    async delegate({ agent, role, repo, cwd, prompt, traceId, taskId }) {
      policy({ agent, role, repo: repo || null, action: "agent.delegate" });
      const adapter = adapters.get(agent);
      const sessionId = newSessionId();
      sess.createSession({ sessionId, taskId, traceId, agent, role,
        tmuxTarget: null, status: "running", startedAt: new Date().toISOString(), closedAt: null });
      try {
        const r = await adapter.delegate({ cwd, prompt, traceId, role, repo });
        sess.setStatus(sessionId, r.exitCode === 0 ? "closed" : "error");
        return { sessionId, ...r };
      } catch (err) {
        sess.setStatus(sessionId, "error");
        auditAppend({ type: "ERROR", traceId, sessionId, where: "agent.delegate", error: String(err.message || err) });
        throw err;
      }
    },

    async spawn({ agent, role, repo, cwd, traceId, taskId }) {
      policy({ agent, role, repo: repo || null, action: "agent.spawn" });
      const adapter = adapters.get(agent);
      const r = await adapter.spawn({ cwd, traceId, role });
      const sessionId = r.sessionId;
      sess.createSession({ sessionId, taskId, traceId, agent, role,
        tmuxTarget: r.tmuxTarget, status: "running", startedAt: new Date().toISOString(), closedAt: null });
      return r;
    },

    async ask({ sessionId, prompt, traceId }) {
      const row = sess.getSession(sessionId);
      if (!row) throw new Error(`unknown session ${sessionId}`);
      const adapter = adapters.get(row.agent);
      return await adapter.ask({ tmuxTarget: row.tmux_target, prompt, traceId, role: row.role });
    },

    async view({ sessionId }) {
      const row = sess.getSession(sessionId);
      if (!row) throw new Error(`unknown session ${sessionId}`);
      return await adapters.get(row.agent).view({ tmuxTarget: row.tmux_target });
    },

    async kill({ sessionId, traceId }) {
      const row = sess.getSession(sessionId);
      if (!row) throw new Error(`unknown session ${sessionId}`);
      const adapter = adapters.get(row.agent);
      const r = await adapter.kill({ tmuxTarget: row.tmux_target, traceId, role: row.role });
      sess.setStatus(sessionId, "closed");
      return r;
    }
  };
}
```

### 3. Test

`tests/gateway/agent_service.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import fs from "node:fs"; import os from "node:os"; import path from "node:path";
import { initState, _resetForTests as resetState } from "../../gateway/src/core/state.js";
import { configureAudit, _resetForTests as resetAudit } from "../../gateway/src/core/audit.js";
import { createAdapterRegistry } from "../../gateway/src/adapters/index.js";
import { createAgentService, PolicyDeniedError } from "../../gateway/src/services/agent_service.js";
import { loadRegistries } from "../../gateway/src/core/registry.js";

function fresh() {
  resetState(); resetAudit();
  initState({ stateDb: path.join(fs.mkdtempSync(path.join(os.tmpdir(),"k-")), "x.db") });
  configureAudit({ auditLog: path.join(fs.mkdtempSync(path.join(os.tmpdir(),"a-")), "audit.jsonl") });
}

function fakeAdapter() {
  return {
    delegate: async () => ({ stdout: "ok", stderr: "", exitCode: 0, dryRun: true }),
    spawn:    async () => ({ sessionId: "ag-x", tmuxTarget: "ag-x", attachCommand: "tmux attach -t ag-x", dryRun: true }),
    ask:      async () => ({ snapshot: "ok", dryRun: true }),
    view:     async () => ({ snapshot: "ok", dryRun: true }),
    kill:     async () => ({ closed: true, dryRun: true }),
  };
}

const reg = loadRegistries({ policiesDir: "policies" });

test("delegate_creates_session_and_result", async () => {
  fresh();
  const adapters = createAdapterRegistry({ config: {}, registries: reg });
  adapters.register("gemini-cli", fakeAdapter());
  const svc = createAgentService({ adapters, registries: reg });
  const r = await svc.delegate({
    agent: "gemini-cli", role: "restricted-coder", repo: "cvision",
    cwd: "/tmp", prompt: "x", traceId: "tr1", taskId: null
  });
  assert.equal(r.exitCode, 0);
  assert.ok(r.sessionId);
});

test("policy_deny_blocks_delegate", async () => {
  fresh();
  const adapters = createAdapterRegistry({ config: {}, registries: reg });
  adapters.register("claude-code", fakeAdapter());
  const svc = createAgentService({ adapters, registries: reg });
  await assert.rejects(() => svc.delegate({
    agent: "claude-code", role: "coder", repo: "cvision",
    cwd: "/tmp", prompt: "x", traceId: "tr2", taskId: null
  }), PolicyDeniedError);
});

test("ask_requires_existing_session", async () => {
  fresh();
  const adapters = createAdapterRegistry({ config: {}, registries: reg });
  const svc = createAgentService({ adapters, registries: reg });
  await assert.rejects(() => svc.ask({ sessionId: "no-such", prompt: "x", traceId: "tr3" }));
});
```

## Errores comunes a evitar

- ❌ Llamar al adapter antes de policy. ADR-003.
- ❌ Persistir session despues del adapter (si adapter throw, no hay session).
- ❌ Olvidar `sessionId` en el resultado.

- [ ] Delegate persiste session y aplica policy.
- [ ] Spawn devuelve `tmuxTarget` y `attachCommand`.
- [ ] Ask/view/kill requieren `sessionId` valido.
- [ ] Policy deny lanza `PolicyDeniedError`.
