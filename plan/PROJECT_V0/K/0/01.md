### 2. `gateway/src/tools/agent.js`

```javascript
import { defineTool, z } from "./tool_helpers.js";

export function buildAgentTools({ agentService }) {
  return [
    defineTool({
      name: "agent.delegate",
      description: "Run an agent in headless mode for a one-shot task",
      schema: z.object({
        agent: z.string(), role: z.string(),
        repo: z.string().optional(),
        cwd: z.string(), prompt: z.string(),
        traceId: z.string(), taskId: z.string().optional()
      }),
      handler: agentService.delegate
    }),
    defineTool({
      name: "agent.spawn",
      description: "Start a persistent (tmux-backed) agent session",
      schema: z.object({
        agent: z.string(), role: z.string(),
        repo: z.string().optional(),
        cwd: z.string(),
        traceId: z.string(), taskId: z.string().optional()
      }),
      handler: agentService.spawn
    }),
    defineTool({
      name: "agent.ask",
      description: "Send a prompt to an existing agent session",
      schema: z.object({ sessionId: z.string(), prompt: z.string(), traceId: z.string() }),
      handler: agentService.ask
    }),
    defineTool({
      name: "agent.view",
      description: "Capture the current pane snapshot of a session",
      schema: z.object({ sessionId: z.string() }),
      handler: agentService.view
    }),
    defineTool({
      name: "agent.kill",
      description: "Close an agent session",
      schema: z.object({ sessionId: z.string(), traceId: z.string() }),
      handler: agentService.kill
    }),
  ];
}
```

### 4. Test

`tests/gateway/tool_agent.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import { buildAgentTools } from "../../gateway/src/tools/agent.js";

const fakeService = {
  delegate: async (a) => ({ sessionId: "ss-x", exitCode: 0, dryRun: true, args: a }),
  spawn:    async () => ({ sessionId: "ss-y", tmuxTarget: "ag-y", attachCommand: "tmux attach -t ag-y" }),
  ask:      async () => ({ snapshot: "ok" }),
  view:     async () => ({ snapshot: "ok" }),
  kill:     async () => ({ closed: true })
};

test("mcp_agent_tools_are_registered", () => {
  const tools = buildAgentTools({ agentService: fakeService });
  const names = tools.map((t) => t.name).sort();
  assert.deepEqual(names, ["agent.ask","agent.delegate","agent.kill","agent.spawn","agent.view"]);
});

test("dry_run_cycle_spawn_ask_view_kill", async () => {
  const tools = buildAgentTools({ agentService: fakeService });
  const [, spawn, ask, view, kill] = tools.sort((a,b)=>a.name.localeCompare(b.name));
  // Order alphabetical: ask,delegate,kill,spawn,view → re-derive
});

test("missing_required_arg_returns_structured_error", async () => {
  const tools = buildAgentTools({ agentService: fakeService });
  const ask = tools.find((t) => t.name === "agent.ask");
  const r = await ask.handler({});
  assert.ok(r.isError);
});
```
