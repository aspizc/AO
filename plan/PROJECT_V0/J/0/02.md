## ¿Por que existe?

Tener services es necesario pero invisible al LLM-orchestrator. El LLM se comunica solo via tools MCP. Aqui exponemos `orchestration.create/view/pause/resume/cancel` y `task.assign`. Sin ellas el orchestrator no puede empezar trabajo.

## ¿Que hay que hacer?

`gateway/src/tools/orchestration.js`, `gateway/src/tools/task.js` con `defineTool`. Cada tool delega al service. Registrarlas en `tools/index.js`.

import { defineTool, z } from "./tool_helpers.js";
import * as svc from "../services/orchestration_service.js";

const CallerSchema = z.object({ agent: z.string(), role: z.string() });

### 3. `gateway/src/tools/task.js`

```javascript
import { defineTool, z } from "./tool_helpers.js";
import { assignTask } from "../services/task_service.js";

export function buildTaskTools({ registries }) {
  return [
    defineTool({
      name: "task.assign",
      description: "Assign work to a target agent in a target role under an orchestration trace",
      schema: z.object({
        traceId: z.string(),
        caller: z.object({ agent: z.string(), role: z.string() }),
        target: z.object({
          agent: z.string().optional(),
          role: z.string(),
          action: z.string().optional()
        }),
        repo: z.string().optional(),
        brief: z.string().optional()
      }),
      handler: (args) => assignTask({ ...args, registries })
    })
  ];
}
```

### 5. Test

`tests/gateway/tool_orchestration_task.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import fs from "node:fs"; import os from "node:os"; import path from "node:path";
import { initState, _resetForTests as resetState } from "../../gateway/src/core/state.js";
import { configureAudit, _resetForTests as resetAudit } from "../../gateway/src/core/audit.js";
import { loadRegistries } from "../../gateway/src/core/registry.js";
import { buildOrchestrationTools } from "../../gateway/src/tools/orchestration.js";
import { buildTaskTools } from "../../gateway/src/tools/task.js";

function fresh() {
  resetState(); resetAudit();
  initState({ stateDb: path.join(fs.mkdtempSync(path.join(os.tmpdir(),"t-")), "x.db") });
  configureAudit({ auditLog: path.join(fs.mkdtempSync(path.join(os.tmpdir(),"a-")), "audit.jsonl") });
}

const reg = loadRegistries({ policiesDir: "policies" });

test("mcp_orchestration_create_and_view", async () => {
  fresh();
  const [createTool, viewTool] = buildOrchestrationTools();
  const r = JSON.parse((await createTool.handler({ callerAgent: "claude-code", callerRole: "orchestrator" })).content[0].text);
  assert.match(r.traceId, /^tr-/);
  const v = JSON.parse((await viewTool.handler({ traceId: r.traceId })).content[0].text);
  assert.ok(v.session);
});

test("mcp_task_assign_via_tool", async () => {
  fresh();
  const [createTool] = buildOrchestrationTools();
  const [assign] = buildTaskTools({ registries: reg });
  const r = JSON.parse((await createTool.handler({ callerAgent: "claude-code", callerRole: "orchestrator" })).content[0].text);
  const t = JSON.parse((await assign.handler({
    traceId: r.traceId,
    caller: { agent: "claude-code", role: "orchestrator" },
    target: { role: "restricted-coder", action: "code.write" },
    repo: "cvision"
  })).content[0].text);
  assert.equal(t.assignedAgent, "gemini-cli");
});

test("invalid_args_return_structured_error", async () => {
  const [createTool] = buildOrchestrationTools();
  const r = await createTool.handler({});
  assert.ok(r.isError);
});
```
