## ¿Por que existe?

`task.assign` es la operacion de orquestacion clave. Aqui es donde el orchestrator descompone trabajo: "Gemini, asume rol restricted-coder y revisa cvision". Mal implementado, abre TM-01 (asignar a si mismo `code.write`) y TM-02 (asignar Claude a `restricted-coder`).

Tambien gestiona la **seleccion automatica** de agente cuando el orchestrator omite `targetAgent`: por ejemplo, repo `restricted` + rol `restricted-coder` → Gemini.

## ¿Que hay que hacer?

`gateway/src/services/task_service.js` con `assignTask({ caller, target, repo, brief })`. Aplica policy a caller y target.

### 2. Service

```javascript
import { evaluate } from "../core/policy_engine.js";
import { newChildTaskId } from "../core/ids.js";
import * as tasks from "../core/repositories/task_repo.js";
import { append as auditAppend } from "../core/audit.js";

export class PolicyDeniedError extends Error {
  constructor(decision) { super(decision.reason); this.code = "POLICY_DENIED"; this.decision = decision; }
}

function selectAgentForRole({ role, repo, registries }) {
  // V4 default: restricted-coder + restricted repo => gemini-cli
  if (role === "restricted-coder") return "gemini-cli";
  // For non-restricted, prefer claude-code if it exists; else gemini-cli
  if (registries.getAgent("claude-code")) return "claude-code";
  return "gemini-cli";
}

export function assignTask({ caller, target, repo, brief, traceId, registries }) {
  // 1. Caller must be orchestrator and authorized for task.assign
  const callerCheck = evaluate({
    agent: caller.agent, role: caller.role,
    action: "task.assign",
    targetAgent: target.agent || null,
    targetRole: target.role
  }, registries);
  if (callerCheck.decision !== "allow") throw new PolicyDeniedError(callerCheck);

  // 2. Resolve target agent if missing
  const targetAgent = target.agent || selectAgentForRole({ role: target.role, repo, registries });

  // 3. Target authorization for the work it will do
  const targetCheck = evaluate({
    agent: targetAgent, role: target.role,
    action: target.action || "code.read",
    repo: repo || null
  }, registries);
  if (targetCheck.decision !== "allow" && targetCheck.decision !== "allow_with_sanitization") {
    throw new PolicyDeniedError(targetCheck);
  }

  const row = {
    taskId: newChildTaskId(),
    traceId,
    assignedAgent: targetAgent,
    assignedRole: target.role,
    repo: repo || null,
    status: "pending",
    createdAt: new Date().toISOString(),
    closedAt: null
  };
  tasks.createTask(row);
  auditAppend({
    type: "TASK_CREATED", traceId, taskId: row.taskId,
    callerAgent: caller.agent, callerRole: caller.role,
    assignedAgent: targetAgent, assignedRole: target.role,
    repo, brief: (brief || "").slice(0, 500)
  });
  return row;
}
```

### 3. Test

`tests/gateway/task_service.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import fs from "node:fs"; import os from "node:os"; import path from "node:path";
import { initState, _resetForTests as resetState } from "../../gateway/src/core/state.js";
import { configureAudit, _resetForTests as resetAudit } from "../../gateway/src/core/audit.js";
import { loadRegistries } from "../../gateway/src/core/registry.js";
import * as orch from "../../gateway/src/services/orchestration_service.js";
import { assignTask, PolicyDeniedError } from "../../gateway/src/services/task_service.js";

function fresh() {
  resetState(); resetAudit();
  initState({ stateDb: path.join(fs.mkdtempSync(path.join(os.tmpdir(),"j-")), "x.db") });
  configureAudit({ auditLog: path.join(fs.mkdtempSync(path.join(os.tmpdir(),"a-")), "audit.jsonl") });
}

const reg = loadRegistries({ policiesDir: "policies" });

test("orchestrator_can_assign_restricted_coder_to_gemini", () => {
  fresh();
  const r = orch.createOrchestration({ callerAgent: "claude-code", callerRole: "orchestrator" });
  const t = assignTask({
    caller: { agent: "claude-code", role: "orchestrator" },
    target: { role: "restricted-coder", action: "code.write" },
    repo: "cvision", brief: "Update X", traceId: r.traceId, registries: reg
  });
  assert.equal(t.assignedAgent, "gemini-cli");
});

test("non_orchestrator_cannot_assign", () => {
  fresh();
  const r = orch.createOrchestration({ callerAgent: "claude-code", callerRole: "orchestrator" });
  assert.throws(() => assignTask({
    caller: { agent: "claude-code", role: "coder" },
    target: { agent: "gemini-cli", role: "coder" },
    repo: "sample-apps", traceId: r.traceId, registries: reg
  }), PolicyDeniedError);
});

test("agent_can_be_selected_when_omitted", () => {
  fresh();
  const r = orch.createOrchestration({ callerAgent: "claude-code", callerRole: "orchestrator" });
  const t = assignTask({
    caller: { agent: "claude-code", role: "orchestrator" },
    target: { role: "coder", action: "code.write" },
    repo: "sample-apps", traceId: r.traceId, registries: reg
  });
  assert.ok(["claude-code","gemini-cli"].includes(t.assignedAgent));
});

test("invalid_target_role_for_target_agent_is_denied", () => {
  fresh();
  const r = orch.createOrchestration({ callerAgent: "claude-code", callerRole: "orchestrator" });
  assert.throws(() => assignTask({
    caller: { agent: "claude-code", role: "orchestrator" },
    target: { agent: "claude-code", role: "restricted-coder", action: "code.write" },
    repo: "cvision", traceId: r.traceId, registries: reg
  }), PolicyDeniedError);
});
```
