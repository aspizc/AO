## ¿Por que existe?

Una orchestration session es la unidad de trabajo del LLM-orchestrator: lo que se crea con `orchestration.create`, contiene `traceId` unico, anida tasks y artifacts, y se cierra con `complete/cancel`. Es el **scope** que correlaciona todo. Sin esto, `task.assign` no tiene a que pertenecer.

## ¿Que hay que hacer?

`gateway/src/services/orchestration_service.js` con `create/view/pause/resume/cancel/complete`. Persiste en `orchestration_sessions` (F/0/2) y emite eventos audit. View agrega tasks/artifacts hijos.

### 2. Service

```javascript
import { newTraceId, newOrchestrationId } from "../core/ids.js";
import * as orch from "../core/repositories/orchestration_repo.js";
import * as tasks from "../core/repositories/task_repo.js";
import * as art from "../core/repositories/artifact_repo.js";
import { append as auditAppend } from "../core/audit.js";

function nowIso() { return new Date().toISOString(); }

export function createOrchestration({ callerAgent, callerRole, goal = null, prefix = null }) {
  if (callerRole !== "orchestrator") {
    const err = new Error("only orchestrator role can create an orchestration");
    err.code = "ROLE_FORBIDDEN"; throw err;
  }
  const traceId = newTraceId({ prefix });
  const sessionId = newOrchestrationId();
  const row = {
    sessionId, traceId, callerAgent, callerRole,
    status: "active", goal, createdAt: nowIso()
  };
  orch.createOrchestration(row);
  auditAppend({ type: "ORCHESTRATION_CREATED", traceId, sessionId, callerAgent, callerRole, goal });
  return row;
}

export function viewOrchestration({ traceId }) {
  const session = orch.getOrchestrationByTraceId(traceId);
  if (!session) return null;
  return {
    session,
    tasks: tasks.listTasksByTrace(traceId),
    artifacts: art.listArtifactsByTrace(traceId)
  };
}

function setStatus(traceId, status, eventType) {
  const session = orch.getOrchestrationByTraceId(traceId);
  if (!session) throw new Error(`unknown traceId ${traceId}`);
  orch.setStatus(session.session_id, status);
  auditAppend({ type: eventType, traceId, sessionId: session.session_id, status });
  return { ...session, status };
}

export const pauseOrchestration  = ({ traceId }) => setStatus(traceId, "paused",   "ORCHESTRATION_PAUSED");
export const resumeOrchestration = ({ traceId }) => setStatus(traceId, "active",   "ORCHESTRATION_RESUMED");
export const cancelOrchestration = ({ traceId }) => setStatus(traceId, "cancelled","ORCHESTRATION_CANCELLED");
export const completeOrchestration = ({ traceId }) => setStatus(traceId, "completed","ORCHESTRATION_COMPLETED");
```

### 3. Test

`tests/gateway/orchestration_service.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import fs from "node:fs"; import os from "node:os"; import path from "node:path";
import { initState, _resetForTests as resetState } from "../../gateway/src/core/state.js";
import { configureAudit, _resetForTests as resetAudit } from "../../gateway/src/core/audit.js";
import * as svc from "../../gateway/src/services/orchestration_service.js";

function fresh() {
  resetState(); resetAudit();
  initState({ stateDb: path.join(fs.mkdtempSync(path.join(os.tmpdir(),"o-")), "x.db") });
  configureAudit({ auditLog: path.join(fs.mkdtempSync(path.join(os.tmpdir(),"a-")), "audit.jsonl") });
}

test("create_persists_trace", () => {
  fresh();
  const r = svc.createOrchestration({ callerAgent: "claude-code", callerRole: "orchestrator", goal: "Refactor X" });
  assert.match(r.traceId, /^tr-/);
  const v = svc.viewOrchestration({ traceId: r.traceId });
  assert.equal(v.session.session_id, r.sessionId);
});

test("non_orchestrator_role_cannot_create", () => {
  fresh();
  assert.throws(() => svc.createOrchestration({ callerAgent: "claude-code", callerRole: "coder" }));
});

test("pause_and_resume_update_status", () => {
  fresh();
  const r = svc.createOrchestration({ callerAgent: "claude-code", callerRole: "orchestrator" });
  svc.pauseOrchestration({ traceId: r.traceId });
  let v = svc.viewOrchestration({ traceId: r.traceId });
  assert.equal(v.session.status, "paused");
  svc.resumeOrchestration({ traceId: r.traceId });
  v = svc.viewOrchestration({ traceId: r.traceId });
  assert.equal(v.session.status, "active");
});

test("cancel_updates_status", () => {
  fresh();
  const r = svc.createOrchestration({ callerAgent: "claude-code", callerRole: "orchestrator" });
  svc.cancelOrchestration({ traceId: r.traceId });
  const v = svc.viewOrchestration({ traceId: r.traceId });
  assert.equal(v.session.status, "cancelled");
});
```
