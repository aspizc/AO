## ¿Por que existe?

Sin un adapter Claude no podemos delegar trabajo a Claude. La interfaz tiene que ser **identica a Gemini** (`delegate/spawn/ask/view/kill`) para que `AgentService` lo use sin diferencias.

## ¿Que hay que hacer?

`gateway/src/adapters/claude_adapter.js`. Estructura espejada de `gemini_adapter.js`. Usa `AGENTS_CLAUDE_BIN` (default `claude`). Modo dry-run por defecto si la flag headless aun no esta confirmada (O/0/0).

### 2. Adapter

```javascript
import { spawnSync } from "node:child_process";
import { BaseAdapter, assertSafeCwd } from "./base_adapter.js";
import {
  isTmuxAvailable, tmuxSync,
  buildNewSessionCmd, buildSendKeysCmd,
  buildCapturePaneCmd, buildKillSessionCmd
} from "./tmux_client.js";
import { buildTmuxTarget } from "./session_naming.js";
import { append as auditAppend } from "../core/audit.js";
import { evaluate } from "../core/policy_engine.js";

function preflight({ registries }, ctx) {
  const r = evaluate(ctx, registries);
  if (r.decision !== "allow") {
    const err = new Error(`policy denied: ${r.reason}`);
    err.policy = r; throw err;
  }
  return r;
}

export class ClaudeAdapter extends BaseAdapter {
  constructor(opts) { super({ id: "claude-code", ...opts }); }

  async delegate({ cwd, prompt, traceId, role, repo }) {
    preflight(this, { agent: "claude-code", role, action: "agent.delegate", repo: repo || null });
    const safeCwd = assertSafeCwd(cwd, this.config.repoRoots);
    auditAppend({ type: "SESSION_STARTED", traceId, agent: "claude-code", role, mode: "headless" });

    if (this.config.dryRun) {
      const r = { stdout: `[dry-run claude]\n${prompt}\ncwd=${safeCwd}`, stderr: "", exitCode: 0, dryRun: true };
      auditAppend({ type: "SESSION_CLOSED", traceId, agent: "claude-code", role, exitCode: 0, mode: "headless" });
      return r;
    }
    // Real headless invocation: TODO until O/0/0 confirms a flag.
    // Until then, fall back to error in non-dry-run real mode.
    auditAppend({ type: "ERROR", traceId, agent: "claude-code", role, where: "delegate",
                  error: "headless mode not confirmed; use spawn/tmux instead" });
    throw new Error("claude-code headless not configured; use agent.spawn (tmux supervised)");
  }

  async spawn({ cwd, traceId, role }) {
    preflight(this, { agent: "claude-code", role, action: "agent.spawn" });
    const safeCwd = assertSafeCwd(cwd, this.config.repoRoots);
    const target = buildTmuxTarget({ traceId, agent: "claude-code", role, prefix: this.config.tmuxPrefix });
    auditAppend({ type: "SESSION_STARTED", traceId, agent: "claude-code", role, mode: "supervised", tmuxTarget: target });
    if (!this.config.dryRun) {
      if (!isTmuxAvailable()) throw new Error("tmux required for claude supervised mode");
      const r = tmuxSync(buildNewSessionCmd({ target, cwd: safeCwd }));
      if (r.status !== 0) throw new Error(`tmux new-session failed: ${r.stderr}`);
      tmuxSync(buildSendKeysCmd({ target, line: this.config.claudeBin || "claude" }));
    }
    return { sessionId: target, tmuxTarget: target,
             attachCommand: `tmux attach -t ${target}`, dryRun: !!this.config.dryRun };
  }

  async ask({ tmuxTarget, prompt, traceId, role }) {
    if (this.config.dryRun) return { snapshot: `[dry-run claude ask]\n${prompt}`, dryRun: true };
    tmuxSync(buildSendKeysCmd({ target: tmuxTarget, line: prompt }));
    await new Promise((r) => setTimeout(r, 1500));
    const cap = tmuxSync(buildCapturePaneCmd({ target: tmuxTarget, lines: 400 }));
    auditAppend({ type: "SESSION_INPUT", traceId, agent: "claude-code", role, tmuxTarget, prompt: prompt.slice(0, 200) });
    return { snapshot: cap.stdout || "", dryRun: false };
  }

  async view({ tmuxTarget }) {
    if (this.config.dryRun) return { snapshot: "[dry-run claude view]", dryRun: true };
    return { snapshot: (tmuxSync(buildCapturePaneCmd({ target: tmuxTarget, lines: 400 })).stdout) || "", dryRun: false };
  }

  async kill({ tmuxTarget, traceId, role }) {
    if (!this.config.dryRun) tmuxSync(buildKillSessionCmd({ target: tmuxTarget }));
    auditAppend({ type: "SESSION_CLOSED", traceId, agent: "claude-code", role, tmuxTarget, mode: "supervised" });
    return { closed: true, dryRun: !!this.config.dryRun };
  }
}
```

### 3. Registrar

En `gateway/src/tools/index.js`, donde se construye el `adapters` registry:

```javascript
adapters.register("claude-code", new ClaudeAdapter({ config, registries }));
```

`tests/gateway/claude_adapter.test.js`: dry-run delegate (rejecta si no allow), dry-run spawn/ask/view/kill cycle.
