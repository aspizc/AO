## ¿Por que existe?

Aunque la policy ya impide a Claude tocar `restricted` (C/0/1), queremos un test **explicito** que confirme que `agent.spawn(repo='cvision', agent='claude-code')` se rechaza ANTES de que tmux haga nada. Defense in depth + evidence binding para U/0/4.

## ¿Que hay que hacer?

Tests de integracion que invocan el adapter via `AgentService` y confirman que repos restricted = deny y non-restricted = allow.

### 2. Test

`tests/gateway/claude_policy.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import fs from "node:fs"; import os from "node:os"; import path from "node:path";
import { initState, _resetForTests as resetState } from "../../gateway/src/core/state.js";
import { configureAudit, _resetForTests as resetAudit, query } from "../../gateway/src/core/audit.js";
import { loadRegistries } from "../../gateway/src/core/registry.js";
import { createAdapterRegistry } from "../../gateway/src/adapters/index.js";
import { ClaudeAdapter } from "../../gateway/src/adapters/claude_adapter.js";
import { createAgentService, PolicyDeniedError } from "../../gateway/src/services/agent_service.js";

function fresh() {
  resetState(); resetAudit();
  initState({ stateDb: path.join(fs.mkdtempSync(path.join(os.tmpdir(),"o-")), "x.db") });
  configureAudit({ auditLog: path.join(fs.mkdtempSync(path.join(os.tmpdir(),"a-")), "audit.jsonl") });
  return { root: fs.realpathSync(fs.mkdtempSync(path.join(os.tmpdir(),"r-"))) };
}

function buildSvc(root) {
  const reg = loadRegistries({ policiesDir: "policies" });
  const adapters = createAdapterRegistry({ config: { dryRun: true, repoRoots: [root] }, registries: reg });
  adapters.register("claude-code", new ClaudeAdapter({ config: { dryRun: true, repoRoots: [root] }, registries: reg }));
  return { svc: createAgentService({ adapters, registries: reg }), reg };
}

test("claude_spawn_on_cvision_denied_before_tmux", async () => {
  const { root } = fresh();
  const { svc } = buildSvc(root);
  await assert.rejects(() => svc.spawn({
    agent: "claude-code", role: "coder", repo: "cvision",
    cwd: root, traceId: "tc1", taskId: null
  }), PolicyDeniedError);
  const events = await query({ traceId: "tc1" });
  assert.ok(!events.some((e) => e.type === "SESSION_STARTED"),
            "SESSION_STARTED should not be emitted when policy denies");
});

test("claude_delegate_on_sample_apps_allowed_dry_run", async () => {
  const { root } = fresh();
  const { svc } = buildSvc(root);
  const r = await svc.delegate({
    agent: "claude-code", role: "coder", repo: "sample-apps",
    cwd: root, prompt: "hi", traceId: "tc2", taskId: null
  });
  // delegate falls back to dry-run mock; in real mode O/0/1 throws because
  // headless flag is not confirmed; both paths are acceptable here.
  assert.ok(r);
});
```
