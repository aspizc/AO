## ¿Por que esta tarea existe?

V4 dice "el Gateway es el unico componente nuevo a construir y corre en Node con MCP stdio". Sin un esqueleto Node ejecutable, las tareas posteriores no tienen donde colgar tools, services, adapters, ni tests Node. Este task levanta el `package.json` y un MCP server stub que arranca, lista cero tools y termina limpiamente.

Es **stub**: nada de logica de negocio, nada de policy, nada de tools reales. Solo:

1. Que arranca.
2. Que lee config.
3. Que NO escribe basura en stdout (porque stdout es para MCP).
4. Que `npm test` corre (aunque no haya tests todavia).

## ¿Que hay que hacer?

Crear `gateway/package.json`, `gateway/src/config.js`, `gateway/src/mcp_server.js` (stub), `gateway/src/tools/index.js` (registry vacio), y un test smoke con `node --test`.

### 3. `gateway/src/config.js`

```javascript
import path from "node:path";
import { fileURLToPath } from "node:url";

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);
const REPO_ROOT = path.resolve(__dirname, "..", "..");

function resolveUnderWorkspace(envValue, defaultRelPath, workspace) {
  if (envValue && path.isAbsolute(envValue)) return envValue;
  const rel = envValue || defaultRelPath;
  return path.resolve(workspace, rel);
}

export function loadConfig(env = process.env) {
  const workspace = env.AGENTS_WORKSPACE
    ? path.resolve(env.AGENTS_WORKSPACE)
    : path.resolve(REPO_ROOT, "workspace");

  return {
    repoRoot: REPO_ROOT,
    workspace,
    policiesDir: env.AGENTS_POLICIES_DIR
      ? path.resolve(env.AGENTS_POLICIES_DIR)
      : path.resolve(REPO_ROOT, "policies"),
    stateDb: resolveUnderWorkspace(env.AGENTS_STATE_DB, "state/state.db", workspace),
    auditLog: resolveUnderWorkspace(env.AGENTS_AUDIT_LOG, "audit/events.jsonl", workspace),
    tmuxPrefix: env.AGENTS_TMUX_PREFIX || "ag-",
    dryRun: env.AGENTS_DRY_RUN === "1"
  };
}
```

### 5. `gateway/src/mcp_server.js`

```javascript
#!/usr/bin/env node
import { loadConfig } from "./config.js";
import { getToolRegistry } from "./tools/index.js";

function logStderr(level, msg, extra = {}) {
  process.stderr.write(JSON.stringify({
    timestamp: new Date().toISOString(),
    level,
    component: "gateway",
    msg,
    ...extra
  }) + "\n");
}

async function main() {
  const config = loadConfig();
  const tools = getToolRegistry();
  logStderr("info", "gateway boot", {
    workspace: config.workspace,
    tools: tools.length
  });

  // MCP wiring is added in G/0/0. For A/0/2 we only confirm the
  // process starts, reads config, and does not write to stdout.
  // Keep stdin open so the process does not exit if used as MCP child;
  // exit immediately if STDIN is a TTY (manual run).
  if (process.stdin.isTTY) {
    logStderr("info", "stub server: TTY detected, exiting cleanly");
    return;
  }
  await new Promise((resolve) => process.stdin.on("end", resolve));
}

main().catch((err) => {
  logStderr("error", "gateway boot failed", { error: String(err) });
  process.exit(1);
});
```

### 6. Test smoke

`gateway/tests/scaffold.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import { loadConfig } from "../src/config.js";
import { getToolRegistry } from "../src/tools/index.js";

test("loadConfig returns required fields", () => {
  const c = loadConfig({ AGENTS_WORKSPACE: "/tmp/aw" });
  assert.equal(c.workspace, "/tmp/aw");
  assert.ok(c.stateDb.includes("state.db"));
  assert.ok(c.auditLog.includes("events.jsonl"));
});

test("loadConfig defaults are local to repo workspace", () => {
  const c = loadConfig({});
  assert.ok(c.workspace.endsWith("/workspace"));
});

test("tool registry is empty in scaffold", () => {
  assert.deepEqual(getToolRegistry(), []);
});
```
