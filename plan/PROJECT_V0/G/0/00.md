## ¿Por que existe?

Hasta ahora el Gateway solo arrancaba como stub. Ahora tiene que ser un **servidor MCP real** capaz de:

- Atender por stdio.
- Listar tools dinamicamente.
- Llevar logs solo a stderr (TM-10).
- Inicializar registries, state y audit.

Sin esto, ninguna tool MCP es accesible por un cliente externo.

## ¿Que hay que hacer?

Cablear `@modelcontextprotocol/sdk` en `gateway/src/mcp_server.js`. Inicializar dependencias de core. Registrar tools del registry (vacio por ahora).

### 2. Reescribir `gateway/src/mcp_server.js`

```javascript
#!/usr/bin/env node
import { Server } from "@modelcontextprotocol/sdk/server/index.js";
import { StdioServerTransport } from "@modelcontextprotocol/sdk/server/stdio.js";
import {
  CallToolRequestSchema,
  ListToolsRequestSchema
} from "@modelcontextprotocol/sdk/types.js";

import { loadConfig } from "./config.js";
import { loadRegistries } from "./core/registry.js";
import { initState } from "./core/state.js";
import { configureAudit, append as auditAppend } from "./core/audit.js";
import { getToolRegistry } from "./tools/index.js";

function logErr(level, msg, extra = {}) {
  process.stderr.write(JSON.stringify({
    ts: new Date().toISOString(), level, component: "gateway", msg, ...extra
  }) + "\n");
}

async function main() {
  const config = loadConfig();
  const registries = loadRegistries({ policiesDir: config.policiesDir });
  initState({ stateDb: config.stateDb });
  configureAudit({ auditLog: config.auditLog });
  auditAppend({ type: "GATEWAY_BOOT", workspace: config.workspace });

  const server = new Server(
    { name: "agents-gateway", version: "0.1.0" },
    { capabilities: { tools: {} } }
  );

  const tools = getToolRegistry({ config, registries });

  server.setRequestHandler(ListToolsRequestSchema, async () => ({
    tools: tools.map((t) => ({
      name: t.name,
      description: t.description,
      inputSchema: t.inputSchema
    }))
  }));

  server.setRequestHandler(CallToolRequestSchema, async (req) => {
    const tool = tools.find((t) => t.name === req.params.name);
    if (!tool) throw new Error(`unknown tool ${req.params.name}`);
    return await tool.handler(req.params.arguments || {});
  });

  const transport = new StdioServerTransport();
  await server.connect(transport);
  logErr("info", "gateway connected", { tools: tools.length });
}

main().catch((err) => {
  logErr("error", "boot failed", { error: String(err), stack: err?.stack });
  process.exit(1);
});
```

### 3. Test smoke

`tests/gateway/mcp_bootstrap.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import { spawn } from "node:child_process";
import { setTimeout as sleep } from "node:timers/promises";
import fs from "node:fs";
import os from "node:os";
import path from "node:path";

test("server_initializes_and_lists_tools", async () => {
  const ws = fs.mkdtempSync(path.join(os.tmpdir(), "ws-"));
  const child = spawn("node", ["gateway/src/mcp_server.js"], {
    env: { ...process.env, AGENTS_WORKSPACE: ws },
    stdio: ["pipe", "pipe", "pipe"]
  });

  let stderr = "";
  child.stderr.on("data", (d) => { stderr += d.toString(); });

  // Send minimal MCP initialize + tools/list (raw JSON-RPC over stdio)
  const requests = [
    { jsonrpc: "2.0", id: 1, method: "initialize",
      params: { protocolVersion: "2024-11-05", capabilities: {}, clientInfo: { name: "smoke", version: "0" } } },
    { jsonrpc: "2.0", id: 2, method: "tools/list" }
  ];
  for (const req of requests) child.stdin.write(JSON.stringify(req) + "\n");

  let out = "";
  child.stdout.on("data", (d) => { out += d.toString(); });
  await sleep(800);
  child.kill("SIGINT");

  assert.ok(stderr.includes("gateway connected"), `stderr did not log connect: ${stderr}`);
  // stdout should contain only protocol JSON, never "gateway connected"
  assert.ok(!out.includes("gateway connected"), "stdout was polluted with logs");
});
```
