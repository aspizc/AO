## ¿Por que existe?

Sin validacion antes del handler, un input malformado de un cliente MCP puede crashear el Gateway o, peor, llegar al policy engine con campos vacios. Necesitamos un helper comun `defineTool` que:

1. Valide con Zod.
2. Convierta errores en `MCP error` (no Stack trace).
3. Centralice el patron para que las 15+ tools posteriores lo reusen.

## ¿Que hay que hacer?

Crear `gateway/src/tools/tool_helpers.js` con `defineTool({ name, description, schema, handler })`. Tests que aseguren errores estructurados.

### 2. `gateway/src/tools/tool_helpers.js`

```javascript
import { z } from "zod";

export class ToolValidationError extends Error {
  constructor(issues) {
    super("invalid input");
    this.name = "ToolValidationError";
    this.issues = issues;
  }
}

function zodToJsonSchema(schema) {
  // Lightweight conversion. The MCP SDK accepts plain JSON Schema; for the
  // subset we use (object, string, number, boolean, enum, optional) this is
  // enough. We avoid the full zod-to-json-schema dependency.
  const shape = schema._def?.shape?.() || {};
  const properties = {};
  const required = [];
  for (const [k, v] of Object.entries(shape)) {
    properties[k] = z2j(v);
    if (!v.isOptional()) required.push(k);
  }
  return { type: "object", properties, required, additionalProperties: false };
}

function z2j(v) {
  if (v._def.typeName === "ZodOptional") return z2j(v._def.innerType);
  if (v._def.typeName === "ZodString") return { type: "string" };
  if (v._def.typeName === "ZodNumber") return { type: "number" };
  if (v._def.typeName === "ZodBoolean") return { type: "boolean" };
  if (v._def.typeName === "ZodEnum") return { type: "string", enum: v._def.values };
  if (v._def.typeName === "ZodArray") return { type: "array", items: z2j(v._def.type) };
  if (v._def.typeName === "ZodObject") return zodToJsonSchema(v);
  return {};
}

export function defineTool({ name, description, schema, handler }) {
  return {
    name,
    description,
    inputSchema: zodToJsonSchema(schema),
    async handler(args) {
      const parsed = schema.safeParse(args);
      if (!parsed.success) {
        const issues = parsed.error.issues.map((i) => ({
          path: i.path.join("."), message: i.message, code: i.code
        }));
        return {
          content: [{ type: "text", text: JSON.stringify({ error: "INVALID_INPUT", issues }) }],
          isError: true
        };
      }
      try {
        const result = await handler(parsed.data);
        return {
          content: [{ type: "text", text: JSON.stringify(result) }]
        };
      } catch (err) {
        return {
          content: [{ type: "text", text: JSON.stringify({ error: "TOOL_ERROR", message: String(err.message || err) }) }],
          isError: true
        };
      }
    }
  };
}

export { z };
```

### 3. Test

`tests/gateway/tool_validation.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import { defineTool, z } from "../../gateway/src/tools/tool_helpers.js";

const tool = defineTool({
  name: "demo.echo",
  description: "Echo for tests",
  schema: z.object({ msg: z.string(), n: z.number().optional() }),
  handler: async ({ msg, n }) => ({ msg, n: n ?? 1 })
});

test("valid_args_pass", async () => {
  const r = await tool.handler({ msg: "hi" });
  const data = JSON.parse(r.content[0].text);
  assert.equal(data.msg, "hi");
  assert.equal(data.n, 1);
});

test("missing_required_arg_returns_mcp_error", async () => {
  const r = await tool.handler({});
  assert.ok(r.isError);
  const data = JSON.parse(r.content[0].text);
  assert.equal(data.error, "INVALID_INPUT");
  assert.ok(data.issues.length > 0);
});

test("handler_exception_does_not_crash_gateway", async () => {
  const t = defineTool({
    name: "demo.boom", description: "x",
    schema: z.object({}), handler: async () => { throw new Error("boom"); }
  });
  const r = await t.handler({});
  assert.ok(r.isError);
});

test("inputSchema_is_present", () => {
  assert.ok(tool.inputSchema);
  assert.equal(tool.inputSchema.type, "object");
  assert.ok(tool.inputSchema.properties.msg);
});
```
