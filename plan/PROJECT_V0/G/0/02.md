## ¿Por que existe?

Primera tool MCP **real**. Reuniendo policy engine + audit + Zod, demuestra el patron extremo a extremo: cliente MCP pide → tool valida → policy decide → audit registra → respuesta con razon. Las 14+ tools posteriores siguen este patron.

## ¿Que hay que hacer?

Crear `gateway/src/tools/policy.js` exportando una tool `policy.check` que llame a `evaluate` y registre `POLICY_DECIDED` en audit. Registrarla en `getToolRegistry()`.

### 2. `gateway/src/tools/policy.js`

```javascript
import { defineTool, z } from "./tool_helpers.js";
import { evaluate, explain } from "../core/policy_engine.js";
import { append as auditAppend } from "../core/audit.js";

const ContextSchema = z.object({
  agent: z.string(),
  role: z.string(),
  action: z.string(),
  repo: z.string().optional(),
  artifactKind: z.string().optional(),
  artifactClassification: z.enum(["unrestricted", "internal", "restricted"]).optional(),
  targetAgent: z.string().optional(),
  targetRole: z.string().optional(),
  targetBranch: z.string().optional(),
  traceId: z.string().optional(),
  explain: z.boolean().optional()
});

export function buildPolicyCheckTool({ registries }) {
  return defineTool({
    name: "policy.check",
    description: "Evaluate the policy engine against a context. Returns decision, reason, ruleId.",
    schema: ContextSchema,
    handler: async (ctx) => {
      const result = ctx.explain
        ? explain(ctx, registries)
        : evaluate(ctx, registries);
      auditAppend({
        type: "POLICY_DECIDED",
        traceId: ctx.traceId || null,
        context: { agent: ctx.agent, role: ctx.role, action: ctx.action, repo: ctx.repo || null },
        decision: result.decision, ruleId: result.ruleId
      });
      return result;
    }
  });
}
```

### 4. Test

`tests/gateway/tool_policy_check.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import fs from "node:fs";
import os from "node:os";
import path from "node:path";

import { loadRegistries } from "../../gateway/src/core/registry.js";
import { configureAudit, query, _resetForTests as resetAudit } from "../../gateway/src/core/audit.js";
import { buildPolicyCheckTool } from "../../gateway/src/tools/policy.js";

const reg = loadRegistries({ policiesDir: "policies" });

function freshAudit() {
  resetAudit();
  const file = path.join(fs.mkdtempSync(path.join(os.tmpdir(), "a-")), "audit.jsonl");
  configureAudit({ auditLog: file });
  return file;
}

test("policy_check_allow", async () => {
  freshAudit();
  const tool = buildPolicyCheckTool({ registries: reg });
  const r = await tool.handler({
    agent: "claude-code", role: "orchestrator", action: "policy.check"
  });
  const data = JSON.parse(r.content[0].text);
  assert.equal(data.decision, "allow");
});

test("policy_check_deny", async () => {
  freshAudit();
  const tool = buildPolicyCheckTool({ registries: reg });
  const r = await tool.handler({
    agent: "claude-code", role: "orchestrator", action: "code.write", repo: "sample-apps"
  });
  const data = JSON.parse(r.content[0].text);
  assert.equal(data.decision, "deny");
});

test("policy_check_writes_audit_event", async () => {
  freshAudit();
  const tool = buildPolicyCheckTool({ registries: reg });
  await tool.handler({
    agent: "claude-code", role: "orchestrator", action: "policy.check", traceId: "t1"
  });
  const events = await query({ traceId: "t1" });
  assert.ok(events.some((e) => e.type === "POLICY_DECIDED"));
});
```

## Errores comunes a evitar
