## ¿Por que esta tarea existe?

Mientras el Gateway escribe eventos, el operador necesita una forma rapida de leerlos sin abrir el fichero a mano. Tambien sirve de utilidad para U/0/0 (E2E) donde verificaras que cada paso del flujo deja `traceId`.

## ¿Que hay que hacer?

Reemplazar el stub `audit show` (de A/0/3) por una implementacion real que invoque al reader Node via subprocess + script `gateway/scripts/query-audit.mjs`. Soporte `--trace-id`, `--type`, `--limit`, `--json`.

### 2. Helper Node

`gateway/scripts/query-audit.mjs`:

```javascript
#!/usr/bin/env node
import path from "node:path";
import { fileURLToPath } from "node:url";
import { configureAudit, query } from "../src/core/audit.js";
import { loadConfig } from "../src/config.js";

const args = process.argv.slice(2);
function get(flag) {
  const i = args.indexOf(flag);
  return i >= 0 ? args[i + 1] : null;
}
const traceId = get("--trace-id");
const type = get("--type");
const limit = Number(get("--limit") || 100);

const config = loadConfig();
configureAudit({ auditLog: config.auditLog });

const events = await query({ traceId, type, limit });
process.stdout.write(JSON.stringify(events));
```

### 3. CLI

En `cli/src/agents_cli/main.py` reemplazar el stub `audit show`:

```python
@audit_app.command("show")
def audit_show(
    trace_id: str = typer.Option(None, "--trace-id"),
    event_type: str = typer.Option(None, "--type"),
    limit: int = typer.Option(50, "--limit"),
    output_json: bool = typer.Option(False, "--json"),
) -> None:
    if shutil.which("node") is None:
        typer.echo("error: node not on PATH", err=True)
        raise typer.Exit(2)
    script = REPO_ROOT / "gateway" / "scripts" / "query-audit.mjs"
    cmd = ["node", str(script), "--limit", str(limit)]
    if trace_id:   cmd += ["--trace-id", trace_id]
    if event_type: cmd += ["--type", event_type]
    proc = subprocess.run(cmd, capture_output=True, text=True, check=False)
    if proc.returncode != 0:
        typer.echo(proc.stderr, err=True)
        raise typer.Exit(proc.returncode)
    events = json.loads(proc.stdout or "[]")
    if output_json:
        typer.echo(json.dumps(events))
        return
    if not events:
        typer.echo("(no events)")
        return
    for e in events:
        if e.get("_corrupt"):
            typer.echo(f"!CORRUPT  {e['raw']}")
            continue
        typer.echo(f"{e['timestamp']}  {e['type']:<24}  trace={e.get('traceId','-'):<16}  {e.get('eventId','')[:8]}")
```

### 4. Test

`tests/cli/test_audit_show.py`:

```python
import json, os, shutil, subprocess
from pathlib import Path
from typer.testing import CliRunner
import pytest

from agents_cli.main import app

REPO = Path(__file__).resolve().parents[2]
runner = CliRunner()


def _seed_audit(tmp_path, events):
    audit = tmp_path / "audit" / "events.jsonl"
    audit.parent.mkdir(parents=True, exist_ok=True)
    for e in events:
        audit.write_text(audit.read_text() if audit.exists() else "" + json.dumps(e) + "\n")
    return audit


def test_audit_show_empty(tmp_path, monkeypatch):
    if shutil.which("node") is None:
        pytest.skip()
    monkeypatch.setenv("AGENTS_WORKSPACE", str(tmp_path))
    res = runner.invoke(app, ["audit", "show"])
    assert res.exit_code == 0
    assert "(no events)" in res.stdout


def test_audit_show_trace_id(tmp_path, monkeypatch):
    if shutil.which("node") is None:
        pytest.skip()
    monkeypatch.setenv("AGENTS_WORKSPACE", str(tmp_path))
    audit = tmp_path / "audit"
    audit.mkdir()
    (audit / "events.jsonl").write_text(
        json.dumps({"eventId": "a", "timestamp": "2026-01-01T00:00:00Z",
                    "type": "X", "traceId": "t1"}) + "\n"
    )
    res = runner.invoke(app, ["audit", "show", "--trace-id", "t1"])
    assert res.exit_code == 0
    assert "t1" in res.stdout


def test_audit_show_type_filter(tmp_path, monkeypatch):
    if shutil.which("node") is None:
        pytest.skip()
    monkeypatch.setenv("AGENTS_WORKSPACE", str(tmp_path))
    audit = tmp_path / "audit"
    audit.mkdir()
    (audit / "events.jsonl").write_text(
        json.dumps({"eventId": "a", "timestamp": "...", "type": "A", "traceId": "t1"}) + "\n" +
        json.dumps({"eventId": "b", "timestamp": "...", "type": "B", "traceId": "t1"}) + "\n"
    )
    res = runner.invoke(app, ["audit", "show", "--type", "B"])
    assert res.exit_code == 0
    assert "B" in res.stdout
    assert "  A  " not in res.stdout  # rough check
```
