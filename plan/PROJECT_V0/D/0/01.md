## ¿Por que esta tarea existe?

Ahora que escribimos eventos, el operador necesita leerlos: por `traceId` (¿que paso en esta orquestacion?), por `type` (¿cuantos `POLICY_DECIDED` hubo?), y con `limit` (no quiero leer 100k lineas). Sin reader, el audit es un agujero sin retorno.

Importante: la corrupcion (linea no parseable) **no se silencia**. Devolver un marker explicito permite que las herramientas lo muestren.

## ¿Que hay que hacer?

Anadir `query({ traceId, type, limit })` a `gateway/src/core/audit.js`. Stream de lectura linea a linea.

### 2. Extender `audit.js`

```javascript
import readline from "node:readline";

export async function query({ traceId, type, limit = 100 } = {}) {
  if (!_config) throw new Error("audit not configured");
  if (!fs.existsSync(_config.auditLog)) return [];
  const stream = fs.createReadStream(_config.auditLog, { encoding: "utf-8" });
  const rl = readline.createInterface({ input: stream });
  const out = [];
  for await (const raw of rl) {
    if (!raw.trim()) continue;
    let evt;
    try {
      evt = JSON.parse(raw);
    } catch (err) {
      out.push({ _corrupt: true, raw });
      continue;
    }
    if (traceId && evt.traceId !== traceId) continue;
    if (type && evt.type !== type) continue;
    out.push(evt);
  }
  if (out.length <= limit) return out;
  return out.slice(out.length - limit);
}
```

### 3. Test

`tests/gateway/audit_reader.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import fs from "node:fs";
import os from "node:os";
import path from "node:path";

import { configureAudit, append, query, _resetForTests } from "../../gateway/src/core/audit.js";

async function setup() {
  _resetForTests();
  const dir = fs.mkdtempSync(path.join(os.tmpdir(), "ar-"));
  const file = path.join(dir, "audit.jsonl");
  configureAudit({ auditLog: file });
  return { dir, file };
}

test("query_by_trace_id", async () => {
  await setup();
  append({ type: "X", traceId: "t1" });
  append({ type: "X", traceId: "t2" });
  const r = await query({ traceId: "t2" });
  assert.equal(r.length, 1);
  assert.equal(r[0].traceId, "t2");
});

test("query_by_type", async () => {
  await setup();
  append({ type: "A", traceId: "t1" });
  append({ type: "B", traceId: "t1" });
  const r = await query({ type: "B" });
  assert.equal(r.length, 1);
});

test("query_limit", async () => {
  await setup();
  for (let i = 0; i < 5; i++) append({ type: "X", traceId: "t" });
  const r = await query({ limit: 2 });
  assert.equal(r.length, 2);
});

test("corrupt_line_is_visible_not_silent", async () => {
  const { file } = await setup();
  fs.appendFileSync(file, "{not json}\n");
  append({ type: "X", traceId: "t" });
  const r = await query({});
  assert.ok(r.some((e) => e._corrupt));
});
```
