## ¿Por que esta tarea existe?

El audit es la **memoria del sistema** y la unica evidencia post-mortem de un incidente. Tiene que ser:

- Append-only (TM-09 / forensics).
- JSONL (una linea = un evento, parseable sin estado).
- Durable (fsync no requerido por evento, pero no se reescriben lineas).

Sin audit, ningun otro stage es seguro: ¿como demostrarias que el sanitizer fallo, o que un approval se otorgo, o que un agente fue denegado? Necesita estar antes que policy/audit-cli/E2E.

## ¿Que hay que hacer?

Crear `gateway/src/core/audit.js` con `append(event)`. Cada evento recibe `eventId` (UUID v4), `timestamp` ISO-8601 UTC, y se escribe como una linea JSON al fichero JSONL configurado.

### 2. `gateway/src/core/audit.js`

```javascript
import fs from "node:fs";
import path from "node:path";
import crypto from "node:crypto";

let _config = null;

export function configureAudit({ auditLog }) {
  if (!auditLog) throw new TypeError("auditLog path required");
  fs.mkdirSync(path.dirname(auditLog), { recursive: true });
  _config = { auditLog };
}

function nowIso() {
  return new Date().toISOString();
}

export function append(event) {
  if (!_config) throw new Error("audit not configured; call configureAudit first");
  if (!event || typeof event !== "object" || !event.type) {
    throw new TypeError("audit event requires {type, ...}");
  }
  const enriched = {
    eventId: crypto.randomUUID(),
    timestamp: nowIso(),
    ...event,
  };
  fs.appendFileSync(_config.auditLog, JSON.stringify(enriched) + "\n", "utf-8");
  return enriched;
}

export function _resetForTests() {
  _config = null;
}
```

### 3. Test

`tests/gateway/audit_writer.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import fs from "node:fs";
import os from "node:os";
import path from "node:path";

import { configureAudit, append, _resetForTests } from "../../gateway/src/core/audit.js";

function tmp() {
  return fs.mkdtempSync(path.join(os.tmpdir(), "audit-"));
}

test("append_creates_file", () => {
  _resetForTests();
  const dir = tmp();
  const file = path.join(dir, "audit.jsonl");
  configureAudit({ auditLog: file });
  append({ type: "TEST_EVENT", traceId: "t1" });
  assert.ok(fs.existsSync(file));
});

test("append_writes_valid_json_line", () => {
  _resetForTests();
  const dir = tmp();
  const file = path.join(dir, "audit.jsonl");
  configureAudit({ auditLog: file });
  const ev = append({ type: "TEST_EVENT", traceId: "t1", payload: { a: 1 } });
  const line = fs.readFileSync(file, "utf-8").trim();
  const parsed = JSON.parse(line);
  assert.equal(parsed.type, "TEST_EVENT");
  assert.equal(parsed.eventId, ev.eventId);
  assert.ok(parsed.timestamp);
});

test("existing_lines_are_not_mutated", () => {
  _resetForTests();
  const dir = tmp();
  const file = path.join(dir, "audit.jsonl");
  configureAudit({ auditLog: file });
  append({ type: "A" });
  const snapshot = fs.readFileSync(file, "utf-8");
  append({ type: "B" });
  const after = fs.readFileSync(file, "utf-8");
  assert.ok(after.startsWith(snapshot), "first line was modified");
  assert.equal(after.split("\n").filter(Boolean).length, 2);
});

test("requires_type_field", () => {
  _resetForTests();
  configureAudit({ auditLog: path.join(tmp(), "x.jsonl") });
  assert.throws(() => append({}), TypeError);
});
```

## Errores comunes a evitar

- ❌ Usar `fs.writeFileSync` (sobreescribe). Usa `appendFileSync`.
- ❌ Permitir eventos sin `type`. Sin tipo, el reader no puede filtrar.
- ❌ Generar `eventId` no-uuid (incrementales adivinables).
- ❌ Olvidar `mkdirSync` con `{recursive:true}`: crash en primera ejecucion.
