## ¿Por que esta tarea existe?

Un humano necesita preguntar al motor "¿que pasaria si Claude intenta esto?" sin levantar el Gateway entero. Esto sirve para:

- Auditoria humana antes de cambiar registries.
- Demos / E2E manuales (U/0/0).
- Habilitar ADR-driven evolution: "ahora el motor decide X" → confirmar con `agent-run policy check`.

## ¿Que hay que hacer?

Implementar el subcomando con todas las flags `--agent`, `--role`, `--repo`, `--action`, `--artifact-kind`, `--artifact-classification`, `--target-agent`, `--target-role`, `--target-branch`, `--json`. Salida `ALLOW`/`DENY`/`REQUIRE_APPROVAL`/`ALLOW_WITH_SANITIZATION` con razon y `ruleId`.

### 2. Helper Node

`gateway/scripts/policy-check.mjs`:

```javascript
#!/usr/bin/env node
import { loadRegistries } from "../src/core/registry.js";
import { evaluate, explain } from "../src/core/policy_engine.js";

const args = process.argv.slice(2);
function get(f) { const i = args.indexOf(f); return i >= 0 ? args[i + 1] : null; }

const ctx = {
  agent:                  get("--agent"),
  role:                   get("--role"),
  repo:                   get("--repo") || null,
  action:                 get("--action"),
  artifactKind:           get("--artifact-kind") || null,
  artifactClassification: get("--artifact-classification") || null,
  targetAgent:            get("--target-agent") || null,
  targetRole:             get("--target-role") || null,
  targetBranch:           get("--target-branch") || null,
};

const reg = loadRegistries({
  policiesDir: process.env.AGENTS_POLICIES_DIR || "policies"
});
const decision = explain(ctx, reg);
process.stdout.write(JSON.stringify(decision));
```

### 3. CLI

```python
@policy_app.command("check")
def policy_check(
    agent: str = typer.Option(...),
    role: str = typer.Option(...),
    action: str = typer.Option(...),
    repo: str = typer.Option(None, "--repo"),
    artifact_kind: str = typer.Option(None, "--artifact-kind"),
    artifact_classification: str = typer.Option(None, "--artifact-classification"),
    target_agent: str = typer.Option(None, "--target-agent"),
    target_role: str = typer.Option(None, "--target-role"),
    target_branch: str = typer.Option(None, "--target-branch"),
    output_json: bool = typer.Option(False, "--json"),
) -> None:
    if shutil.which("node") is None:
        raise typer.Exit(2)
    script = REPO_ROOT / "gateway" / "scripts" / "policy-check.mjs"
    cmd = ["node", str(script),
           "--agent", agent, "--role", role, "--action", action]
    for flag, val in [
        ("--repo", repo), ("--artifact-kind", artifact_kind),
        ("--artifact-classification", artifact_classification),
        ("--target-agent", target_agent), ("--target-role", target_role),
        ("--target-branch", target_branch)
    ]:
        if val: cmd += [flag, val]
    proc = subprocess.run(cmd, capture_output=True, text=True, check=False)
    data = json.loads(proc.stdout)
    if output_json:
        typer.echo(json.dumps(data))
    else:
        d = data["decision"].upper()
        color = {"ALLOW": "green", "DENY": "red",
                 "REQUIRE_APPROVAL": "yellow",
                 "ALLOW_WITH_SANITIZATION": "cyan"}.get(d, "white")
        console.print(f"[{color}]{d}[/{color}] ruleId={data['ruleId']}  reason={data['reason']}")
    raise typer.Exit(0 if data["decision"] == "allow" else 1)
```

### 4. Test

`tests/cli/test_policy_check.py`:

```python
import json, shutil
from typer.testing import CliRunner
import pytest

from agents_cli.main import app

runner = CliRunner()


def _check(*extra):
    if shutil.which("node") is None:
        pytest.skip()
    return runner.invoke(app, ["policy", "check", *extra, "--json"])


def test_canonical_raw_restricted_denied():
    r = _check("--agent", "claude-code", "--role", "orchestrator",
               "--action", "artifact.get",
               "--artifact-kind", "raw_diff",
               "--artifact-classification", "restricted")
    assert r.exit_code != 0
    data = json.loads(r.stdout)
    assert data["decision"] == "deny"


def test_canonical_task_assign_allowed():
    r = _check("--agent", "claude-code", "--role", "orchestrator",
               "--action", "task.assign",
               "--target-agent", "gemini-cli",
               "--target-role", "restricted-coder")
    assert r.exit_code == 0


def test_protected_push_requires_approval():
    r = _check("--agent", "gemini-cli", "--role", "coder",
               "--repo", "cvision",
               "--action", "git.push", "--target-branch", "main")
    data = json.loads(r.stdout)
    assert data["decision"] == "require_approval"
```
