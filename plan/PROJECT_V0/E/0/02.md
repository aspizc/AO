## ¿Por que esta tarea existe?

Hasta ahora cada comando imprimia su propio formato. Para usabilidad y para que un script externo pueda parsear automatizado, queremos: (a) tablas Rich consistentes en modo humano, (b) `--json` en TODOS los comandos relevantes, (c) errores siempre a stderr con exit no-cero.

### 2. `cli/src/agents_cli/output.py`

```python
import json as _json
import sys

import typer
from rich.console import Console
from rich.table import Table

console = Console()
err_console = Console(stderr=True)


def emit(data, *, json_mode: bool, table_renderer=None) -> None:
    if json_mode:
        typer.echo(_json.dumps(data))
        return
    if table_renderer is not None:
        table_renderer(data)
        return
    typer.echo(_json.dumps(data, indent=2))


def fail(msg: str, *, exit_code: int = 1) -> None:
    err_console.print(f"[red]error:[/red] {msg}")
    raise typer.Exit(exit_code)


def render_audit_table(events):
    table = Table(show_header=True, header_style="bold")
    table.add_column("ts")
    table.add_column("type")
    table.add_column("traceId")
    table.add_column("eventId")
    for e in events:
        if e.get("_corrupt"):
            table.add_row("-", "[red]CORRUPT[/red]", "-", "-")
            continue
        table.add_row(e["timestamp"], e["type"],
                      e.get("traceId", "-"),
                      (e.get("eventId", "") or "")[:8])
    console.print(table)


def render_decision(d):
    color = {
        "allow": "green", "deny": "red",
        "require_approval": "yellow",
        "allow_with_sanitization": "cyan",
    }.get(d["decision"], "white")
    console.print(f"[{color}]{d['decision'].upper()}[/{color}]"
                  f"  ruleId={d['ruleId']}  reason={d['reason']}")
```

### 4. Tests

`tests/cli/test_cli_output.py`:

```python
import json
import shutil
from typer.testing import CliRunner
import pytest

from agents_cli.main import app

runner = CliRunner()


def test_policy_check_table_contains_decision():
    if shutil.which("node") is None:
        pytest.skip()
    r = runner.invoke(app, [
        "policy", "check",
        "--agent", "claude-code", "--role", "orchestrator",
        "--action", "policy.check"
    ])
    assert r.exit_code == 0
    assert "ALLOW" in r.stdout


def test_json_mode_outputs_valid_json():
    if shutil.which("node") is None:
        pytest.skip()
    r = runner.invoke(app, ["policy", "validate", "--json"])
    assert r.exit_code == 0
    json.loads(r.stdout)


def test_errors_go_to_stderr():
    if shutil.which("node") is None:
        pytest.skip()
    r = runner.invoke(app, ["policy", "validate", "--policies-dir", "/no/such"])
    assert r.exit_code != 0
    assert r.stderr or "FAIL" in r.output
```
