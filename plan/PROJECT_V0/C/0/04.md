## ¿Por que esta tarea existe?

V4 introduce `allow_with_sanitization` para resolver: ¿que pasa cuando un coder restricted produce `raw_diff` y un reviewer lo necesita? Si bloqueamos, no puede haber review; si dejamos pasar raw, leak. Solucion: el motor responde `allow_with_sanitization`, el servicio M-layer consulta el sanitizer, y solo el sanitized cruza la frontera.

Esta capa **decide** la frontera. Los _rules_ y la implementacion del sanitizer viven en M.

## ¿Que hay que hacer?

Anadir `evaluateSanitization(ctx, registries)`:

- Si `action == artifact.get` y `artifactKind` es raw + `artifactClassification == restricted` y requester carece de raw access → `allow_with_sanitization`.
- Si rol prohibe incluso sanitized (e.g. orchestrator no puede leer `raw_*`) → `deny`.

### 2. Extender engine

```javascript
import { isRawKind } from "./policy_types.js";

function roleHasRawRestrictedAccess(role) {
  if (!role) return false;
  const allow = role.allowActions || [];
  return allow.includes("code.read.raw_restricted") ||
         allow.includes("artifact.put.raw_restricted") ||
         allow.includes("artifact.get.raw_restricted");
}

function evaluateSanitization(ctx, registries) {
  if (ctx.action !== "artifact.get") return null;
  if (!isRawKind(ctx.artifactKind)) return null;

  if (ctx.artifactClassification !== "restricted") return null;

  const role = registries.getRole(ctx.role);
  if (roleHasRawRestrictedAccess(role)) {
    return null; // allowed direct; classification check above already passed
  }

  const denyKinds = role?.denyActions || [];
  if (denyKinds.includes("artifact.get.raw_restricted")) {
    return { decision: Decision.DENY,
             reason: `role ${ctx.role} cannot consume raw restricted even after sanitization`,
             ruleId: "sanitization.role_denies_raw" };
  }

  return { decision: Decision.ALLOW_WITH_SANITIZATION,
           reason: `role ${ctx.role} may consume sanitized version of ${ctx.artifactKind}`,
           ruleId: "sanitization.required" };
}
```

### 3. Test

`tests/gateway/policy_sanitization.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import { loadRegistries } from "../../gateway/src/core/registry.js";
import { evaluate } from "../../gateway/src/core/policy_engine.js";

const reg = loadRegistries({ policiesDir: "policies" });

test("restricted_raw_diff_to_reviewer_requires_sanitization", () => {
  const r = evaluate({
    agent: "claude-code", role: "reviewer",
    repo: "sample-apps", action: "artifact.get",
    artifactKind: "raw_diff", artifactClassification: "restricted"
  }, reg);
  assert.equal(r.decision, "allow_with_sanitization");
});

test("sanitized_diff_to_reviewer_allowed", () => {
  const r = evaluate({
    agent: "claude-code", role: "reviewer",
    repo: "sample-apps", action: "artifact.get",
    artifactKind: "review_notes", artifactClassification: "restricted"
  }, reg);
  // not a raw kind → sanitization layer returns null; previous layers allow
  assert.equal(r.decision, "allow");
});

test("raw_restricted_to_orchestrator_denied", () => {
  const r = evaluate({
    agent: "claude-code", role: "orchestrator",
    repo: null, action: "artifact.get",
    artifactKind: "raw_diff", artifactClassification: "restricted"
  }, reg);
  assert.equal(r.decision, "deny");
});

test("non_restricted_raw_no_sanitization_needed", () => {
  const r = evaluate({
    agent: "claude-code", role: "reviewer",
    repo: "sample-apps", action: "artifact.get",
    artifactKind: "raw_diff", artifactClassification: "internal"
  }, reg);
  // role still denies it (reviewer.denyActions includes raw_restricted only)
  // here the kind is raw_diff but classification is internal → check that
  // previous layers' role rule rejects raw_* in any classification.
  // Asserts non-allow:
  assert.notEqual(r.decision, "allow_with_sanitization");
});
```

## Errores comunes a evitar

- ❌ Tratar `allow_with_sanitization` como un `allow`. Es **un contrato** con el caller: "ALLOW solo si pasas por sanitizer".
- ❌ Perder el case del orchestrator: aunque la sanitizacion pudiera generar un sanitized, el rol orchestrator NO lo recibe (nivel rol).
- ❌ Olvidar que la frontera es `restricted`: artefactos `internal` no requieren sanitizacion.
