## ¿Por que esta tarea existe?

Una de las decisiones explicitas del operador (V4 §13.2): **acciones rutinarias no requieren aprobacion** (commit local, lint, format, test) pero **acciones irreversibles si** (push a `main`, install nuevas deps, deploy). Sin esta capa, o pedimos aprobacion para todo (fatiga) o no la pedimos para nada (peligro).

## ¿Que hay que hacer?

Anadir un layer `evaluateApproval` que devuelve `require_approval` cuando:

1. La accion esta en `agent.requiresApprovalFor`.
2. El target branch hace match con `protectedBranches` (caso `git.push` → `git.push.protected`).
3. La accion es `dependency.change`.

Acciones rutinarias (`code.write` en non-protected, commit local, lint, test) no piden nada.

### 2. Extender engine

```javascript
function matchesProtectedBranch(branch, patterns) {
  if (!branch) return false;
  return patterns.some((p) =>
    p === branch ||
    (p.endsWith("/*") && branch.startsWith(p.slice(0, -1)))
  );
}

function evaluateApproval(ctx, registries) {
  const agent = registries.getAgent(ctx.agent);
  const requires = agent.requiresApprovalFor || [];
  const protectedBranches = registries.getProtectedBranches();

  if (ctx.action === "git.push" && matchesProtectedBranch(ctx.targetBranch, protectedBranches)) {
    return { decision: Decision.REQUIRE_APPROVAL,
             reason: `push to protected branch ${ctx.targetBranch}`,
             ruleId: "approval.git_push_protected" };
  }

  if (requires.includes(ctx.action)) {
    return { decision: Decision.REQUIRE_APPROVAL,
             reason: `agent ${ctx.agent} requires approval for ${ctx.action}`,
             ruleId: "approval.agent_required" };
  }

  return null;
}
```

Anadir `evaluateApproval` al pipeline de `evaluate(...)` despues de `evaluateRole`.

### 3. Test

`tests/gateway/policy_approval.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import { loadRegistries } from "../../gateway/src/core/registry.js";
import { evaluate } from "../../gateway/src/core/policy_engine.js";

const reg = loadRegistries({ policiesDir: "policies" });

test("protected_branch_push_requires_approval", () => {
  const r = evaluate({
    agent: "gemini-cli", role: "coder", repo: "cvision",
    action: "git.push", targetBranch: "main"
  }, reg);
  assert.equal(r.decision, "require_approval");
});

test("dependency_change_requires_approval", () => {
  const r = evaluate({
    agent: "claude-code", role: "coder", repo: "sample-apps",
    action: "dependency.change"
  }, reg);
  assert.equal(r.decision, "require_approval");
});

test("local_commit_is_allowed", () => {
  const r = evaluate({
    agent: "claude-code", role: "coder", repo: "sample-apps",
    action: "code.write"
  }, reg);
  assert.equal(r.decision, "allow");
});

test("test_run_is_allowed", () => {
  const r = evaluate({
    agent: "claude-code", role: "tester", repo: "sample-apps",
    action: "test.run"
  }, reg);
  assert.equal(r.decision, "allow");
});

test("push_to_feature_branch_does_not_require_approval", () => {
  const r = evaluate({
    agent: "gemini-cli", role: "coder", repo: "cvision",
    action: "git.push", targetBranch: "feature/foo"
  }, reg);
  assert.notEqual(r.decision, "require_approval");
});
```
