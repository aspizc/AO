## ¿Por que esta tarea existe?

La frontera mas critica del sistema es la clasificacion del repo. Aqui es donde TM-02 (leak raw restricted) y TM-03 (filesystem bypass) se cierran al nivel de policy. Implementar esto **antes** de roles, approvals y sanitization fuerza a que el resto del motor herede default-deny.

## ¿Que hay que hacer?

Crear `gateway/src/core/policy_engine.js` con `evaluate(context, registries)`. Implementar **solo** las reglas de clasificacion: agente vs clasificacion, repo vs agentes permitidos, paths excluidos.

### 2. `gateway/src/core/policy_engine.js`

```javascript
import { Decision, normalizePolicyContext } from "./policy_types.js";

function deny(reason, ruleId) {
  return { decision: Decision.DENY, reason, ruleId };
}
function allow(reason, ruleId) {
  return { decision: Decision.ALLOW, reason, ruleId };
}

export function evaluate(rawCtx, registries) {
  const ctx = normalizePolicyContext(rawCtx);
  const agent = registries.getAgent(ctx.agent);
  if (!agent) return deny(`unknown agent ${ctx.agent}`, "agent.unknown");

  if (ctx.repo) {
    const repo = registries.getRepo(ctx.repo);
    if (!repo) return deny(`unknown repo ${ctx.repo}`, "repo.unknown");
    if (!agent.allowedClassifications.includes(repo.classification)) {
      return deny(
        `agent ${ctx.agent} cannot operate on classification ${repo.classification}`,
        "classification.agent_not_allowed");
    }
    if (!repo.allowedAgents.includes(ctx.agent)) {
      return deny(
        `repo ${ctx.repo} does not allow agent ${ctx.agent}`,
        "classification.repo_not_allowed");
    }
    if (Array.isArray(repo.excludedPaths) && ctx.path) {
      const hit = repo.excludedPaths.some((p) => ctx.path.startsWith(p));
      if (hit) return deny(`path ${ctx.path} is excluded`, "classification.excluded_path");
    }
  }

  return allow("classification check passed", "classification.ok");
}
```

> Nota: este es solo el primer plano. Stages C/0/2..4 anaden capas. Cada nueva capa devuelve antes de llegar a `allow` final.

### 3. Test

`tests/gateway/policy_classification.test.js`:

```javascript
import { test } from "node:test";
import assert from "node:assert/strict";
import { loadRegistries } from "../../gateway/src/core/registry.js";
import { evaluate } from "../../gateway/src/core/policy_engine.js";

const reg = loadRegistries({ policiesDir: "policies" });

test("claude_denied_on_restricted_repo", () => {
  const r = evaluate({
    agent: "claude-code", role: "coder", repo: "cvision", action: "code.read",
  }, reg);
  assert.equal(r.decision, "deny");
  assert.match(r.ruleId, /classification/);
});

test("codex_denied_on_restricted_repo", () => {
  const r = evaluate({
    agent: "codex", role: "coder", repo: "cvision", action: "code.read",
  }, reg);
  assert.equal(r.decision, "deny");
});

test("gemini_allowed_on_restricted", () => {
  const r = evaluate({
    agent: "gemini-cli", role: "restricted-coder", repo: "cvision", action: "code.read",
  }, reg);
  assert.equal(r.decision, "allow");
});

test("unknown_repo_is_denied", () => {
  const r = evaluate({
    agent: "claude-code", role: "coder", repo: "no-such-repo", action: "code.read",
  }, reg);
  assert.equal(r.decision, "deny");
});

test("excluded_path_is_denied", () => {
  // Synthetic registries, not from file, to test excludedPaths.
  const fakeReg = {
    getAgent: () => ({ allowedClassifications: ["unrestricted"], allowedRoles: ["coder"] }),
    getRepo:  () => ({
      classification: "unrestricted",
      allowedAgents: ["x"], excludedPaths: ["secrets/"],
    }),
    getRole: () => ({}),
    getProtectedBranches: () => []
  };
  const r = evaluate({
    agent: "x", role: "coder", repo: "y", action: "code.read", path: "secrets/key.pem"
  }, fakeReg);
  assert.equal(r.decision, "deny");
});
```
