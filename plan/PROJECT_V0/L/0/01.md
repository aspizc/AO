```javascript
import { defineTool, z } from "./tool_helpers.js";
import * as store from "../core/artifact_store.js";

const PutSchema = z.object({
  traceId: z.string(),
  kind: z.string(),
  classification: z.enum(["unrestricted","internal","restricted"]),
  producedBy: z.string(),
  content: z.string()
});

const GetSchema = z.object({ artifactId: z.string(),
  requesterAgent: z.string().optional(),
  requesterRole: z.string().optional() });

const ListSchema = z.object({ traceId: z.string() });

export function buildArtifactTools() {
  return [
    defineTool({
      name: "artifact.put", description: "Persist an artifact under a traceId",
      schema: PutSchema,
      handler: async (a) => store.put({ ...a, content: Buffer.from(a.content, "utf-8") })
    }),
    defineTool({
      name: "artifact.get", description: "Read an artifact by artifactId",
      schema: GetSchema,
      handler: async ({ artifactId }) => {
        const r = store.get({ artifactId });
        if (!r) return { error: "NOT_FOUND" };
        return { ...r, content: r.content.toString("utf-8") };
      }
    }),
    defineTool({
      name: "artifact.list", description: "List artifacts under a traceId",
      schema: ListSchema,
      handler: async ({ traceId }) => store.list({ traceId })
    })
  ];
}
```

> Nota: `artifact.get` aqui no aplica policy aun. L/0/2 lo anade. La razon: aislar la responsabilidad del store de la responsabilidad de la policy.
